### 轮转公式
**前两个用的多一些，你要看清数组是从下标1开始的，还是从下标0开始的**
- 0基索引下的轮转公式：==新位置 = (当前位置 + 转动次数) % n==(n为一共有多少位数)，举例：0 1 2 3  ,从2开始往后数3个数，新位置应该为1。用公式：新位置=(2+3)%4=1。

- 1基索引下的轮转公式：==新位置 = (当前位置 + 转动次数-1) % n +1 (n为一共有多少位数)==，举例： 1 2 3 4 ,从2开始往后数3个数，新位置应该为1。用公式：新位置=(2+3-1)%4+1=1。

- 2基索引下的轮转公式：==新位置 = (当前位置 + 转动次数 - 2) % n + 2==(n为一共有多少位数)，举例： 2 3 4 5 ,从3开始往后数5个数，新位置应该为4。用公式：新位置=(3+5-2)%4+2=4。

- 剩下的以此类推


[蓝桥杯2025年第十六届省赛真题-抽奖 - C语言网](https://www.dotcpp.com/oj/problem3341.html)

### 题目简介
- 一道简单的模拟题目，难点在于看懂题目的意思

### 代码实现
```cpp
#include <iostream>
#include <vector>
#include <algorithm>  // 用于排序函数sort

using namespace std;

int main() {
    // 1. 输入转轮大小n和抽奖次数m
    int n, m;
    cin >> n >> m;

    // 2. 输入三个转轮的图案（下标从1开始，方便对应题目中的位置）
    vector<int> wheelA(n + 1);  // 转轮A
    vector<int> wheelB(n + 1);  // 转轮B
    vector<int> wheelC(n + 1);  // 转轮C

    for (int i = 1; i <= n; ++i) {
        cin >> wheelA[i];
    }
    for (int i = 1; i <= n; ++i) {
        cin >> wheelB[i];
    }
    for (int i = 1; i <= n; ++i) {
        cin >> wheelC[i];
    }

    // 3. 初始化三个转轮的初始位置（都在第1格）和总积分
    int posA = 1, posB = 1, posC = 1;
    int totalScore = 0;

    // 4. 循环处理m次抽奖
    for (int i = 0; i < m; ++i) {
        // 输入本次每个转轮的转动次数
        int x1, x2, x3;
        cin >> x1 >> x2 >> x3;

        // 更新转轮位置：核心公式 (当前位置 + 转动次数 - 1) % n + 1
        posA = (posA + x1 - 1) % n + 1;
        posB = (posB + x2 - 1) % n + 1;
        posC = (posC + x3 - 1) % n + 1;

        // 获取当前三个转轮指向的数值
        int a = wheelA[posA];
        int b = wheelB[posB];
        int c = wheelC[posC];

        // 5. 按照优先级判断本次积分
        int addScore = 0;
        // 优先级1：三个相同 或 顺序连续 → +200分
        if ((a == b && b == c) || (a + 1 == b && b + 1 == c)) {
            addScore = 200;
        }
        // 优先级2：不满足200分的情况下，判断是否满足100分条件
        else {
            // 条件1：有两个数值相同
            bool hasTwoSame = (a == b) || (a == c) || (b == c);
            // 条件2：三个数排序后连续
            int temp[] = { a, b, c };
            sort(temp, temp + 3);  // 对三个数排序
            bool sortedConsec = (temp[0] + 1 == temp[1] && temp[1] + 1 == temp[2]);

            if (hasTwoSame || sortedConsec) {
                addScore = 100;
            }
        }

        // 累加本次积分到总分
        totalScore += addScore;
    }

    // 6. 输出最终总积分
    cout << totalScore << endl;

    return 0;
}
```
