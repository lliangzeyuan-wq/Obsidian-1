```cpp
#include <iostream>
#include <string>
#include <unordered_map>
#include <queue>
#include <functional>
#include <fstream>
#include <vector>
#include <iomanip>
using namespace std;

using uint = unsigned int;

class HuffmanTree
{
public:
    HuffmanTree()
        : minHeap_([](Node* n1, Node* n2) { return n1->weight_ > n2->weight_; })
        , root_(nullptr)
    {
    } 

    ~HuffmanTree()
    {
        destroy(root_);
    }

    // 功能I：初始化 - 读入字符和权值，建立哈夫曼树，存入文件hfmTree
    void initFromInput()
    {
        destroy(root_);
        root_ = nullptr;
        while (!minHeap_.empty()) minHeap_.pop();
        codeMap_.clear();

        int n;
        cout << "请输入字符集大小n: ";
        cin >> n;
        vector<pair<char, uint>> chars(n);
        for (int i = 0; i < n; ++i)
        {
            cout << "请输入第" << i + 1 << "个字符和权值（格式：字符 权值）: ";
            cin >> chars[i].first >> chars[i].second;
        }

        // 生成节点放入小根堆
        for (auto& p : chars)
        {
            minHeap_.push(new Node(p.first, p.second));
        }

        buildTree();

        // 存入文件hfmTree
        saveTreeToFile("hfmTree");
        cout << "--- 哈夫曼树已建立并保存到hfmTree文件\n";
    }

    // 从文件hfmTree读取哈夫曼树
    bool loadTreeFromFile(const string& filename = "hfmTree")
    {
        ifstream ifs(filename, ios::binary);
        if (!ifs.is_open()) return false;

        destroy(root_);
        root_ = nullptr;
        while (!minHeap_.empty()) minHeap_.pop();
        codeMap_.clear();

        // 读取叶子节点信息（简化存储格式：字符+权值）
        int leafCount;
        ifs.read((char*)&leafCount, sizeof(int));
        vector<pair<char, uint>> leaves(leafCount);
        for (int i = 0; i < leafCount; ++i)
        {
            ifs.read((char*)&leaves[i].first, sizeof(char));
            ifs.read((char*)&leaves[i].second, sizeof(uint));
        }

        for (auto& p : leaves)
        {
            minHeap_.push(new Node(p.first, p.second));
        }
        buildTree();
        return true;
    }

    // 功能E：编码 - 对ToBeTran.txt文件编码，结果存入CodeFile
    bool encodeFile(const string& inFile = "ToBeTran.txt", const string& outFile = "CodeFile")
    {
        if (!root_)
        {
            if (!loadTreeFromFile())
            {
                cout << "--- 错误：哈夫曼树未建立，请先执行I命令\n";
                return false;
            }
        }
        getHuffmanCode();

        ifstream ifs(inFile);
        if (!ifs.is_open())
        {
            cout << "--- 错误：无法打开文件" << inFile << endl;
            return false;
        }
        string text((istreambuf_iterator<char>(ifs)), istreambuf_iterator<char>());
        ifs.close();

        string codeStr;
        for (char ch : text)
        {
            if (codeMap_.find(ch) == codeMap_.end())
            {
                cout << "--- 错误：字符 '" << ch << "' 不在字符集中\n";
                return false;
            }
            codeStr += codeMap_[ch];
        }

        ofstream ofs(outFile);
        if (!ofs.is_open())
        {
            cout << "--- 错误：无法创建文件" << outFile << endl;
            return false;
        }
        ofs << codeStr;
        ofs.close();
        cout << "--- 编码完成，结果已保存到" << outFile << endl;
        return true;
    }

    // 功能D：译码 - 对CodeFile文件译码，结果存入TextFile
    bool decodeFile(const string& inFile = "CodeFile", const string& outFile = "TextFile")
    {
        if (!root_)
        {
            if (!loadTreeFromFile())
            {
                cout << "--- 错误：哈夫曼树未建立，请先执行I命令\n";
                return false;
            }
        }

        ifstream ifs(inFile);
        if (!ifs.is_open())
        {
            cout << "--- 错误：无法打开文件" << inFile << endl;
            return false;
        }
        string codeStr((istreambuf_iterator<char>(ifs)), istreambuf_iterator<char>());
        ifs.close();

        string text;
        Node* cur = root_;
        for (char ch : codeStr)
        {
            if (ch == '0') cur = cur->left_;
            else cur = cur->right_;

            if (!cur->left_ && !cur->right_)
            {
                text += cur->data_;
                cur = root_;
            }
        }

        ofstream ofs(outFile);
        if (!ofs.is_open())
        {
            cout << "--- 错误：无法创建文件" << outFile << endl;
            return false;
        }
        ofs << text;
        ofs.close();
        cout << "--- 译码完成，结果已保存到" << outFile << endl;
        return true;
    }

    // 功能P：印代码文件 - 每行50个代码，同时写入CodePrin文件
    bool printCodeFile(const string& inFile = "CodeFile", const string& outFile = "CodePrin")
    {
        ifstream ifs(inFile);
        if (!ifs.is_open())
        {
            cout << "--- 错误：无法打开文件" << inFile << endl;
            return false;
        }
        string codeStr((istreambuf_iterator<char>(ifs)), istreambuf_iterator<char>());
        ifs.close();

        ofstream ofs(outFile);
        if (!ofs.is_open())
        {
            cout << "--- 错误：无法创建文件" << outFile << endl;
            return false;
        }

        cout << "\n--- 代码文件内容（每行50个）---\n";
        for (size_t i = 0; i < codeStr.size(); i += 50)
        {
            string line = codeStr.substr(i, 50);
            cout << line << endl;
            ofs << line << endl;
        }
        ofs.close();
        cout << "--- 代码文件已打印并保存到" << outFile << endl;
        return true;
    }

    // 功能T：印哈夫曼树 - 树形/缩进形式显示，写入TreePrint文件
    bool printTree(const string& filename = "TreePrint")
    {
        if (!root_)
        {
            cout << "--- 错误：哈夫曼树未建立，请先执行I命令\n";
            return false;
        }

        ofstream ofs(filename);
        if (!ofs.is_open())
        {
            cout << "--- 错误：无法创建文件" << filename << endl;
            return false;
        }

        cout << "\n--- 哈夫曼树（缩进形式）---\n";
        printTreeHelper(root_, 0, cout);
        printTreeHelper(root_, 0, ofs);
        ofs.close();
        cout << "--- 哈夫曼树已打印并保存到" << filename << endl;
        return true;
    }

private:
    struct Node
    {
        Node(char data, uint weight)
            : data_(data), weight_(weight), left_(nullptr), right_(nullptr) {
        }
        char data_;
        uint weight_;
        Node* left_;
        Node* right_;
    };

    Node* root_;
    unordered_map<char, string> codeMap_;
    using MinHeap = priority_queue<Node*, vector<Node*>, function<bool(Node*, Node*)>>;
    MinHeap minHeap_;

    void destroy(Node* node)
    {
        if (!node) return;
        destroy(node->left_);
        destroy(node->right_);
        delete node;
    }

    void buildTree()
    {
        while (minHeap_.size() > 1)
        {
            Node* n1 = minHeap_.top(); minHeap_.pop();
            Node* n2 = minHeap_.top(); minHeap_.pop();
            Node* parent = new Node('\0', n1->weight_ + n2->weight_);
            parent->left_ = n1;
            parent->right_ = n2;
            minHeap_.push(parent);
        }
        if (!minHeap_.empty())
        {
            root_ = minHeap_.top();
            minHeap_.pop();
        }
    }

    void getHuffmanCode()
    {
        codeMap_.clear();
        string code;
        getHuffmanCode(root_, code);
    }

    void getHuffmanCode(Node* node, string code)
    {
        if (!node->left_ && !node->right_)
        {
            if (code.empty()) code = "0";
            codeMap_[node->data_] = code;
            return;
        }
        if (node->left_) getHuffmanCode(node->left_, code + "0");
        if (node->right_) getHuffmanCode(node->right_, code + "1");
    }

    void saveTreeToFile(const string& filename)
    {
        // 收集所有叶子节点
        vector<pair<char, uint>> leaves;
        collectLeaves(root_, leaves);

        ofstream ofs(filename, ios::binary);
        int leafCount = leaves.size();
        ofs.write((char*)&leafCount, sizeof(int));
        for (auto& p : leaves)
        {
            ofs.write((char*)&p.first, sizeof(char));
            ofs.write((char*)&p.second, sizeof(uint));
        }
        ofs.close();
    }

    void collectLeaves(Node* node, vector<pair<char, uint>>& leaves)
    {
        if (!node) return;
        if (!node->left_ && !node->right_)
        {
            leaves.emplace_back(node->data_, node->weight_);
            return;
        }
        collectLeaves(node->left_, leaves);
        collectLeaves(node->right_, leaves);
    }

    void printTreeHelper(Node* node, int level, ostream& os)
    {
        if (!node) return;
        os << setw(level * 4) << "";
        if (!node->left_ && !node->right_)
            os << "Leaf: '" << node->data_ << "' (weight: " << node->weight_ << ")\n";
        else
            os << "Internal Node (weight: " << node->weight_ << ")\n";
        printTreeHelper(node->left_, level + 1, os);
        printTreeHelper(node->right_, level + 1, os);
    }
};

// 菜单主函数
int main()
{
    HuffmanTree htree;
    char op;
    do
    {
        cout << "\n===== 哈夫曼编/译码系统菜单 =====\n";
        cout << "I - 初始化哈夫曼树（读入字符和权值）\n";
        cout << "E - 编码文件 ToBeTran.txt -> CodeFile\n";
        cout << "D - 译码文件 CodeFile -> TextFile\n";
        cout << "P - 打印代码文件（每行50个）\n";
        cout << "T - 打印哈夫曼树\n";
        cout << "Q - 退出\n";
        cout << "请输入选择：";
        cin >> op;
        op = toupper(op);
        
        switch (op)
        {
        case 'I':
            htree.initFromInput();
            break;
        case 'E':
            htree.encodeFile();
            break;
        case 'D':
            htree.decodeFile();
            break;
        case 'P':
            htree.printCodeFile();
            break;
        case 'T':
            htree.printTree();
            break;
        case 'Q':
            cout << "退出系统...\n";
            break;
        default:
            cout << "无效输入，请重新选择\n";
        }
    } while (op != 'Q');

    return 0;
}
```
