## 模块 5：哈夫曼树构建核心算法

**作用**：根据输入的字符和权值，构建出哈夫曼树。

**开发顺序**：**第五个写**，这是**整个系统的心脏**，没有它就没有后续的编解码。

cpp

运行

```
public:
    // 构建哈夫曼树
    void buildTree()
    {
        // 只要堆里还有超过1个节点，就继续合并
        while (minHeap_.size() > 1)
        {
            // 第一步：取出两个权值最小的节点
            Node* n1 = minHeap_.top(); minHeap_.pop();
            Node* n2 = minHeap_.top(); minHeap_.pop();
            
            // 第二步：生成父节点，权值为两个子节点之和
            Node* parent = new Node('\0', n1->weight_ + n2->weight_);
            parent->left_ = n1;
            parent->right_ = n2;
            
            // 第三步：把父节点放回堆中，继续下一轮合并
            minHeap_.push(parent);
        }
        // 最后堆里剩下的那个节点，就是整棵树的根节点
        if (!minHeap_.empty())
        {
            root_ = minHeap_.top();
            minHeap_.pop();
        }
    }
```

## 模块 6：编码功能模块

**作用**：根据哈夫曼树生成每个字符的编码，然后对文件进行编码。

**开发顺序**：**第六个写**，有了树才能生成编码表，有了编码表才能编码。

cpp

运行

```
private:
    // 辅助函数：递归遍历树，生成每个字符的哈夫曼编码
    void getHuffmanCode(Node* node, string code)
    {
        // 到达叶子节点，说明找到了一个字符的编码
        if (!node->left_ && !node->right_)
        {
            // 特殊情况：只有一个字符时，编码为"0"
            if (code.empty()) code = "0";
            // 把字符和对应的编码存入映射表
            codeMap_[node->data_] = code;
            return;
        }
        // 左子树走0，右子树走1
        if (node->left_) getHuffmanCode(node->left_, code + "0");
        if (node->right_) getHuffmanCode(node->right_, code + "1");
    }

public:
    // 对外接口：生成编码表
    void getHuffmanCode()
    {
        codeMap_.clear();
        string code;
        getHuffmanCode(root_, code);
    }

    // 功能E：文件编码
    bool encodeFile(const string& inFile = "ToBeTran.txt", const string& outFile = "CodeFile")
    {
        // 如果内存中没有树，尝试从文件加载
        if (!root_)
        {
            if (!loadTreeFromFile())
            {
                cout << "❌ 错误：哈夫曼树未建立，请先执行I命令\n";
                return false;
            }
        }
        // 先生成编码表
        getHuffmanCode();

        // 1. 读取待编码的文本文件
        ifstream ifs(inFile);
        if (!ifs.is_open())
        {
            cout << "❌ 错误：无法打开文件" << inFile << endl;
            return false;
        }
        //把文件里面的所有内容读出来，存到text中
        string text((istreambuf_iterator<char>(ifs)), istreambuf_iterator<char>());
        ifs.close();

        // 2. 根据编码表，把文本转换成编码串
        string codeStr;
        for (char ch : text)
        {
            if (codeMap_.find(ch) == codeMap_.end())
            {
                cout << "❌ 错误：字符 '" << ch << "' 不在字符集中\n";
                return false;
            }
            codeStr += codeMap_[ch];
        }

        // 3. 把编码串写入输出文件
        ofstream ofs(outFile);
        if (!ofs.is_open())
        {
            cout << "❌ 错误：无法创建文件" << outFile << endl;
            return false;
        }
        ofs << codeStr;
        ofs.close();
        cout << "✅ 编码完成，结果已保存到" << outFile << endl;
        return true;
    }
```

运行

public:
    // 功能D：文件解码
    bool decodeFile(const string& inFile = "CodeFile", const string& outFile = "TextFile")
    {
        // 如果内存中没有树，尝试从文件加载
        if (!root_)
        {
            if (!loadTreeFromFile())
            {
                cout << "❌ 错误：哈夫曼树未建立，请先执行I命令\n";
                return false;
            }
        }

        // 1. 读取编码文件
        ifstream ifs(inFile);
        if (!ifs.is_open())
        {
            cout << "❌ 错误：无法打开文件" << inFile << endl;
            return false;
        }
        string codeStr((istreambuf_iterator<char>(ifs)), istreambuf_iterator<char>());
        ifs.close();

        // 2. 遍历编码串，还原原始文本
        string text;
        Node* cur = root_; // 从根节点开始遍历
        for (char ch : codeStr)
        {
            // 遇到0走左子树，遇到1走右子树
            if (ch == '0') cur = cur->left_;
            else cur = cur->right_;

            // 到达叶子节点，说明找到了一个字符
            if (!cur->left_ && !cur->right_)
            {
                text += cur->data_;
                cur = root_; // 回到根节点，准备解码下一个字符
            }
        }

        // 3. 把还原的文本写入输出文件
        ofstream ofs(outFile);
        if (!ofs.is_open())
        {
            cout << "❌ 错误：无法创建文件" << outFile << endl;
            return false;
        }
        ofs << text;
        ofs.close();
        cout << "✅ 译码完成，结果已保存到" << outFile << endl;
        return true;
    }
```

## 模块 8：树的持久化模块

**作用**：把哈夫曼树保存到文件，下次运行程序可以直接加载，不用重新输入字符和权值。

**开发顺序**：**第八个写**，提升用户体验，避免重复输入。

cpp

运行

```
private:
    // 辅助函数：收集所有叶子节点（用于保存树）
    void collectLeaves(Node* node, vector<pair<char, uint>>& leaves)
    {
        if (!node) return;
        if (!node->left_ && !node->right_)
        {
            leaves.emplace_back(node->data_, node->weight_);
            return;
        }
        collectLeaves(node->left_, leaves);
        collectLeaves(node->right_, leaves);
    }

    // 把哈夫曼树保存到二进制文件
    void saveTreeToFile(const string& filename)
    {
        vector<pair<char, uint>> leaves;
        collectLeaves(root_, leaves);

        // 二进制方式写入文件
        ofstream ofs(filename, ios::binary);
        int leafCount = leaves.size();
        ofs.write((char*)&leafCount, sizeof(int));
        for (auto& p : leaves)
        {
            ofs.write((char*)&p.first, sizeof(char));
            ofs.write((char*)&p.second, sizeof(uint));
        }
        ofs.close();
    }

public:
    // 从二进制文件加载哈夫曼树
    bool loadTreeFromFile(const string& filename = "hfmTree")
    {
        ifstream ifs(filename, ios::binary);
        if (!ifs.is_open()) return false;

        // 先清空之前的树
        destroy(root_);
        root_ = nullptr;
        while (!minHeap_.empty()) minHeap_.pop();
        codeMap_.clear();

        // 读取叶子节点信息
        int leafCount;
        ifs.read((char*)&leafCount, sizeof(int));
        vector<pair<char, uint>> leaves(leafCount);
        for (int i = 0; i < leafCount; ++i)
        {
            ifs.read((char*)&leaves[i].first, sizeof(char));
            ifs.read((char*)&leaves[i].second, sizeof(uint));
        }

        // 重新构建哈夫曼树
        for (auto& p : leaves)
        {
            minHeap_.push(new Node(p.first, p.second));
        }
        buildTree();
        return true;
    }

    // 功能I：用户输入初始化
    void initFromInput()
    {
        // 先清空之前的树
        destroy(root_);
        root_ = nullptr;
        while (!minHeap_.empty()) minHeap_.pop();
        codeMap_.clear();

        // 从终端读入字符和权值
        int n;
        cout << "请输入字符集大小n: ";
        cin >> n;
        vector<pair<char, uint>> chars(n);
        for (int i = 0; i < n; ++i)
        {
            cout << "请输入第" << i + 1 << "个字符和权值（格式：字符 权值）: ";
            cin >> chars[i].first >> chars[i].second;
        }

        // 生成节点放入小根堆
        for (auto& p : chars)
        {
            minHeap_.push(new Node(p.first, p.second));
        }

        // 构建树并保存到文件
        buildTree();
        saveTreeToFile("hfmTree");
        cout << "✅ 哈夫曼树已建立并保存到hfmTree文件\n";
    }
```

运行

```
private:
    // 辅助函数：递归打印哈夫曼树（缩进形式）
    void printTreeHelper(Node* node, int level, ostream& os)
    {
        if (!node) return;
        // 根据层级输出缩进
        os << setw(level * 4) << "";
        if (!node->left_ && !node->right_)
            os << "Leaf: '" << node->data_ << "' (weight: " << node->weight_ << ")\n";
        else
            os << "Internal Node (weight: " << node->weight_ << ")\n";
        // 递归打印左右子树
        printTreeHelper(node->left_, level + 1, os);
        printTreeHelper(node->right_, level + 1, os);
    }

public:
    // 功能P：打印代码文件（每行50个）
    bool printCodeFile(const string& inFile = "CodeFile", const string& outFile = "CodePrin")
    {
        ifstream ifs(inFile);
        if (!ifs.is_open())
        {
            cout << "❌ 错误：无法打开文件" << inFile << endl;
            return false;
        }
        string codeStr((istreambuf_iterator<char>(ifs)), istreambuf_iterator<char>());
        ifs.close();

        ofstream ofs(outFile);
        if (!ofs.is_open())
        {
            cout << "❌ 错误：无法创建文件" << outFile << endl;
            return false;
        }

        cout << "\n--- 代码文件内容（每行50个）---\n";
        for (size_t i = 0; i < codeStr.size(); i += 50)
        {
            string line = codeStr.substr(i, 50);
            cout << line << endl;
            ofs << line << endl;
        }
        ofs.close();
        cout << "✅ 代码文件已打印并保存到" << outFile << endl;
        return true;
    }

    // 功能T：打印哈夫曼树
    bool printTree(const string& filename = "TreePrint")
    {
        if (!root_)
        {
            cout << "❌ 错误：哈夫曼树未建立，请先执行I命令\n";
            return false;
        }

        ofstream ofs(filename);
        if (!ofs.is_open())
        {
            cout << "❌ 错误：无法创建文件" << filename << endl;
            return false;
        }

        cout << "\n--- 哈夫曼树（缩进形式）---\n";
        printTreeHelper(root_, 0, cout);
        printTreeHelper(root_, 0, ofs);
        ofs.close();
        cout << "✅ 哈夫曼树已打印并保存到" << filename << endl;
        return true;
    }
}; // 哈夫曼树类结束
