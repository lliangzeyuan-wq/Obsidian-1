public:
	//不带权值的最短路径问题 - 广度优先遍历
	void shortPath(int start , int end)
	{
		vector<bool>visited(vertics.size(), false);
		queue<int>que;
		//记录顶点在遍历过程中的前后遍历关系
		vector<int>path(vertics.size(), 0);

		que.push(start);
		visited[start] = true;
		while (!que.empty())
		{
			int cur_no = que.front();
			//找到end末尾节点
			if (cur_no == end)
			{
				break;
			}
			que.pop();
			for (auto no : vertics[cur_no].adjList_)
			{
				if (!visited[no])
				{
					que.push(no);
					visited[no] = true;
					//当前节点处，记录是从哪一个节点过来的
					path[no] = cur_no; 
				}
			}
		}
		//cout << endl;

		if (!que.empty() )
		{
			////存在一条最短路径
			//while (end != 0)
			//{
			//	cout << vertics[end].data_ << " <= ";
			//	end = path[end];
			//}

			showPath(end,path);
			
		}
		else
		{
			cout << "不存在有效的最短路径" << endl;
		}
		cout << endl;
	}
	
private:
	//输出最短路径信息
	//利用递归的回溯性质实现倒序输出
	void showPath(int end, vector<int>& path)
	{
		if (end == 0)
		{
			return;
		}
		
		showPath(path[end], path);
		cout << vertics[end].data_ << " ";
	}
