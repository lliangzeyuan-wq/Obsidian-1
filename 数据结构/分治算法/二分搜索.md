- 递归实现

```cpp
#include<iostream>
using namespace std;
#include<cstdlib> //rand()
#include<vector>
#include<algorithm> //sort()
bool binarySearch(vector<int>& vec, int i, int j, int val)
{
	//元素不存在
	if (i > j)
	{
		return false;
	}
	int mid = (i + j) / 2;
	if (vec[mid] == val)
	{
		return true;
	}
	else if (vec[mid] > val)
	{
		return binarySearch(vec, i, mid - 1, val);
	}
	else
	{
		return binarySearch(vec, mid + 1, j, val);
	}
}
int main()
{
	vector<int>vec;
	for (int i = 0; i < 11; ++i)
	{
		vec.push_back(rand() % 100);
	}
	sort(vec.begin(), vec.end());
	for (int v : vec)
	{
		cout << v << " ";
	}
	cout << endl;

	bool result = binarySearch(vec, 0, vec.size() - 1,50);
	cout << "result:" << result << endl;
	return 0;
}
```
