### 效率低的写法
- 效率低下的原因：每一次往上面“归”的时候，都要重新计算左边和右边的高度，重复计算了
```cpp
public:
	//判断平衡树
	bool isBalance() {
		return isBalance(root_);
	}
private:
	//判断平衡树的代码实现
	bool isBalance(Node* node) {
		if (node == NULL) {
			return true;
		}
		if (!isBalance(node->left_)) {//L
			return false;
		}
		if (!isBalance(node->right_)) {//R
			return false;
		}
		int left = high(node->left_);
		int right = high(node->right_);
		return abs(left - right) <= 1;//V
	}
```


### 优化后的写法
```cpp
public:
	//判断平衡树
	bool isBalance() {
		int l = 0;
		bool flag = true;
		isBalance02(root_, l, flag);
		return flag;
	}
private:
	//判断平衡树 效率高 递归过程中，记录了节点的高度值  返回节点高度值
	int isBalance02(Node*node,int l,bool&flag) {
		if (node == NULL) {
			return l;
		}

		//一定是要递归到底才能知道到底你的左右子树是不是平衡树
		int left = isBalance02(node->left_, l + 1, flag);//L
		//因为你flag已经判断过了啊，你要的就是flag是true还是false,直接返回就行了
		if (!flag) {
			return left;
		}

		int right = isBalance02(node->right_, l + 1, flag);//R
		if (!flag) {
			return right;
		}

		//V
		if (abs(left - right) > 1) {
			flag = false;
		}
		return max(left, right);
	}
```
