### 效率低的写法
- 效率低下的原因：每一次往上面“归”的时候，都要重新计算左边和右边的高度，重复计算了
```cpp
public:
	//判断平衡树
	bool isBalance() {
		return isBalance(root_);
	}
private:
	//判断平衡树的代码实现
	bool isBalance(Node* node) {
		if (node == NULL) {
			return true;
		}
		if (!isBalance(node->left_)) {//L
			return false;
		}
		if (!isBalance(node->right_)) {//R
			return false;
		}
		int left = high(node->left_);
		int right = high(node->right_);
		return abs(left - right) <= 1;//V
	}
```
