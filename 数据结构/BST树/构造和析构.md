```cpp
public:
	//构造函数
	BSTree(Comp comp=Comp())
		:root_(nullptr)
		,comp_(comp)
	{}
	//层序遍历思想释放BST树所有节点资源
	~BSTree(){
		if (root_ != nullptr) {
			queue<Node*>s;
			s.push(root_);
			while(!s.empty()) {
				Node* front = s.front();
				s.pop();

				if (front->left_ != nullptr) {
					s.push(front->left_);
				}
				if (front->right_ != nullptr) {
					s.push(front->right_);
				}
				delete front;
			}
		}
	}
```

- 析构：每删除一个节点，都把不为空的子节点推入队列中
