==红黑树的删除最多旋转3次==
- 一.如果删除的是一个红色节点，不影响，不用做任何删除调整操作
- 二.如果删除的是一个黑色节点，分两种情况
	- 1.如果补上来的孩子是红色节点，直接把这个孩子涂成黑色，调整完成！！！
	- 2. 如果补上来的孩子是黑色节点，那在当前这个路径上没办法再补一个黑色节点出来了，只能从兄弟那里借调黑色节点了
### 借调的情况（二、2）
###### 情况1
- A为红或黑，兄弟节点的孩子为红色
- ==C：右红==
![[Pasted image 20260209180756.png]]
###### 情况2
- 先以兄弟节点为轴做一个右旋转，然后把C、D的颜色互换一下，就转换成情况1了
- ==C：左红==
![[Pasted image 20260209181400.png]]
###### 情况3
- 把C涂成红色，这样左右两边的黑色节点就一样了。然后如果A为红色节点，把A涂成黑色节点；如果A为黑色节点，就算了，这样左右两边的黑色节点数都减1
- ==C：左右都不红==
![[Pasted image 20260209183259.png]]
###### 情况4
- 处理完之后有了一个兄弟节点D，又归结成了上面的三种情况
- C是红的，那么A一定是黑的
- ==C本身红==
![[Pasted image 20260209190445.png]]

public:
	//删除代码
	void remove(const T&val) {
		Node* cur = root_;
		Node* parent = NULL;
		while (cur != NULL) {
			if (cur->data_ < val) {
				parent = cur;
				cur = cur->right_;
			}
			else if (cur->data_ > val) {
				parent = cur;
				cur = cur->left_;
			}
			else {
				return;
			}
		}
		if (cur == NULL) {
			return;
		}
		//开始删除
		if (cur->left_ != nullptr && cur->right_ != nullptr) {
			//找前驱节点
			Node* pre = cur->left_;
			while (pre->right_ != nullptr) {
				pre = pre->right_;
			}
			cur->data_ = pre->data_;
			cur = pre;
		}

		Node* child = cur->left_;
		if (child == nullptr) {
			child = cur->right_;
		}

		if (child != NULL) {
			child->parent_ = cur->parent_;
			if (cur == root_) {
				root_ = child;
			}
			else {
				if (cur->parent_->left_ == cur) {
					cur->parent_->left_ = child;

				}
				else {
					cur->parent_->right_ = child;
				}
			}
			Color c = color(cur);
			delete cur;

			if (c == BLACK)//删除的是黑色节点，要进行删除调整操作
			{
				fixAfterRemove(child);
			}
		}
		//cur没有孩子
		else {
			//child == nullptr
			//删除的cur是根节点
			if (cur->parent_ == NULL) {
				delete cur;
				root_ = nullptr;
				return;
			}
			//删除的cur就是叶子节点
			else {
				if (color(cur) == BLACK) {
					fixAfterRemove(cur);
				}

				if (cur->parent_->left_ == cur) {
					cur->parent_->left_ = NULL;
				}
				else {
					cur->parent_->right_ = NULL;
				}
				delete cur;
			}
		}
	}
	
private:	
	//红黑树的删除调整操作
	void fixAfterRemove(Node* node) {
		while (color(node) == BLACK) {
			//删除的黑色节点在左子树
			if (left(parent(node)) == node) {
				Node* brother = right(parent(node));
				//情况4
				if (color(brother) == RED) {
					setColor(brother, BLACK);
					setColor(parent(node), RED);
					leftRotate(parent(node));
					brother = right(parent(node)); 
				}
				//情况3
				if (color(brother->left_) == BLACK
					&& color(brother->right_) == BLACK) {
					setColor(brother, RED);
					node = parent(node);//对，因为让循环向上传递，能够跳出循环
					//setColor(parent(node), BLACK);  //错，因为形成了死循环
				}
				else {
					//情况2
					if (color(left(brother)) == RED)
					{
						setColor(brother, RED);
						setColor(left(brother), BLACK);
						rightRotate(brother);
						brother = right(parent(node));
					}

					//情况1
					setColor(brother, color(parent(node)));
					setColor(parent(node), BLACK);
					setColor(right(brother), BLACK);
					leftRotate(parent(node));
					break;
				}
			}
			//删除的节点在右子树
			else {
				Node* brother = left(parent(node));
				//情况4
				if (color(brother) == RED) {
					setColor(brother, BLACK);
					setColor(parent(node), RED);
					rightRotate(parent(node));
					brother = left(parent(node));
				}
				//情况3
				if (color(brother->left_) == BLACK
					&& color(brother->right_) == BLACK) {
					setColor(brother, RED);
					node = parent(node);//对，因为让循环向上传递，能够跳出循环
					//setColor(parent(node), BLACK);  //错，因为形成了死循环
				}
				else {
					//情况2
					if (color(right(brother)) == RED)
					{
						setColor(brother, RED);
						setColor(right(brother), BLACK);
						leftRotate(brother);
						brother = left(parent(node));
					}

					//情况1
					setColor(brother, color(parent(node)));
					setColor(parent(node), BLACK);
					setColor(left(brother), BLACK);
					rightRotate(parent(node));
					break;
				}
			}
		}
		//如果发现删除后补上来的child节点是红色，直接涂成黑色，调整结束
		setColor(node, BLACK);
	}
