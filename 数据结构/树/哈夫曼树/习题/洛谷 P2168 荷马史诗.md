# 虚拟节点

因为一次合并k个节点 ，然后生成1一个节点，一共n个节点
那么进行一次： n- k+1 =>>  n-(k-1)   
进行m次： n - m* (k-1)   ，要保证这个式子最终为一 ，即：
`n - m* (k-1)  = 1 `   =>>    `n-1 = m* (k-1)`  =>>  
`(n-1) % (k-1) == 0`   

如果 `(n-1) % (k-1) == r != 0`  我们需要加上 -r+(k-1)  
>n-1 = m* (k-1) + r  ;   => n-1-r = m* (k-1)   
>但是我们不能加上-r这个负数个节点，因此要加上 -r+(k-1)  个虚拟节点


```cpp
#include<iostream>
#include<queue>
#include<algorithm>
using namespace std;

using ll = long long;

struct Node
{
	ll w_;
	ll dep_;
	Node(ll w,ll dep)
		:w_(w)
		,dep_(dep)
	{ }
	bool operator>(const Node& n) const
	{
		if (w_ != n.w_)
		{
			return w_ > n.w_;
		}
		return dep_ > n.dep_;
	}
};

int main()
{
	ll n, k;
	cin >> n >> k;
	priority_queue<Node, vector<Node>, greater<Node>>q;
	for (ll i = 0; i < n; ++i)
	{
		ll w;
		cin >> w;
		q.emplace(w, 0);
	}
	//虚拟节点
	if (k > 2)
	{
		ll mod = (n - 1) % (k - 1);
		if (mod != 0) 
		{
			ll add = (k - 1) - mod;
			for (ll i = 0; i < add; ++i)
			{
				q.emplace(0, 0);
			}
		}

	}

	ll total = 0;//WPL
	ll max_len = 0;//最大深度
	while (q.size() > 1)
	{
		ll sum_w = 0;
		ll max_dep = 0;
		
		for (ll i = 0; i < k; ++i)
		{
			Node cur = q.top(); q.pop();
			sum_w += cur.w_;
			max_dep = max(max_dep, cur.dep_);
		}
		total += sum_w;
		ll new_dep = max_dep + 1;
		max_len = max(max_len, new_dep);
		q.emplace(sum_w, new_dep);
	}

	cout << total << "\n" << max_len << endl;
	return 0;
}
```
