[P1090 [NOIP 2004 提高组] 合并果子 - 洛谷](https://www.luogu.com.cn/problem/P1090)


```cpp
#include<iostream>
#include<queue>
using namespace std;
int main()
{
	int n;
	cin >> n;
	priority_queue < int, vector<int>, greater<int>>q;
	for (int i = 0; i < n; ++i)
	{
		int a;
		cin >> a;
		q.push(a);
	}
	int ans = 0;
	while (q.size() > 1)
	{
		int x = q.top();
		q.pop();
		int y = q.top();
		q.pop();
		int sum = x + y;
		ans += sum;
		q.push(sum);
	}
	cout << ans << endl;
	return 0;
}
```
