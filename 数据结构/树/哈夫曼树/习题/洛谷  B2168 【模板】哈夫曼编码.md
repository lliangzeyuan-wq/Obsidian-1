```cpp
#include<iostream>
#include<string>
#include<unordered_map>
#include<queue>
#include<functional>
using namespace std;

using ll = long long;

class HuffmanTree
{
public:
	HuffmanTree()
		:root_(nullptr),
		minHeap_([](Node* n1, Node* n2)->bool
			{return n1->weight_ > n2->weight_;	})
	{ }
	~HuffmanTree()
	{
		destroy(root_);
	}
	
	void create(const vector<pair<string, ll>>& data)
	{
		
		while (!minHeap_.empty())
		{
			minHeap_.pop();
		}
		codeMap_.clear();

		for (auto& p : data)
		{
			minHeap_.push(new Node(p.first, p.second));
		}

		while (minHeap_.size() > 1)
		{
			Node* n1 = minHeap_.top();
			minHeap_.pop();
			Node* n2 = minHeap_.top();
			minHeap_.pop();
			
			Node* node = new Node("", n1->weight_ + n2->weight_);
			node->left_ = n1;
			node->right_ = n2;

			minHeap_.push(node);
		}
		if (!minHeap_.empty())
		{
			root_ = minHeap_.top();
			minHeap_.pop();
		}
	}

	// 获取并返回每个单词的哈夫曼编码（按输入顺序）
	vector<pair<string,string>> getCodes(const vector<pair<string, ll>>& data)
	{
		string code;
		getHuffmanCode(root_, code);
		
		vector<pair<string, string>>res;
		for (auto& p : data)
		{
			res.emplace_back(p.first, codeMap_[p.first]);
		}
		return res;
	}

private:
	struct Node
	{
		Node(string data,ll weight)
			:data_(data)
			,weight_(weight)
			,left_(nullptr)
			,right_(nullptr)
		{ }
		string data_;
		ll weight_;
		Node* left_;
		Node* right_;
	};


	void destroy(Node*node)
	{
		//LRV
		if (node == nullptr)
		{
			return;
		}
		destroy(node->left_);
		destroy(node->right_);
		
		delete node;
		node = nullptr;
	}


	void getHuffmanCode(Node* node, string code)
	{
		if (node->left_ == nullptr && node->right_ == nullptr)
		{
			if (code.empty()) code = '0';
			codeMap_[node->data_] = code;
			return;
		}
		getHuffmanCode(node->left_, code + '0');
		getHuffmanCode(node->right_, code + '1');
	}
	
private:
	Node* root_;
	unordered_map<string, string>codeMap_;
	using MinHeap = priority_queue<Node*, vector<Node*>, function<bool(Node*, Node*)>>;
	MinHeap minHeap_;
};

int main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	int n;
	cin >> n;

	vector<pair<string, ll>> data(n);
	for (int i = 0; i < n; ++i)
	{
		cin >> data[i].first >> data[i].second;
	}

	HuffmanTree htree;
	htree.create(data);
	auto codes = htree.getCodes(data);

	for (auto& p : codes)
	{
		cout << p.first << " " << p.second << "\n";
	}

	return 0;
}
```
