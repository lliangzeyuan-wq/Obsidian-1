[P1334 瑞瑞的木板 - 洛谷](https://www.luogu.com.cn/problem/P1334)

```cpp
#include<iostream>
#include<queue>
using namespace std;

using ll = long long;

int main()
{
	ll n;
	cin >> n;
	priority_queue<ll, vector<ll>, greater<ll>>q;
	for (ll i = 0; i < n; ++i)
	{
		ll w;
		cin >> w;
		q.emplace(w);
	}
	ll sum = 0;
	while (q.size() > 1)
	{
		ll n1 = q.top();
		q.pop();
		ll n2 = q.top();
		q.pop();
		ll sum_1 = n1 + n2;
		sum += sum_1;
		q.emplace(sum_1);
	}
	cout << sum << endl;
	return 0;
}
```
