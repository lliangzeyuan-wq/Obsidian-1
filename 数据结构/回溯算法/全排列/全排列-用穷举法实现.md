 - 这种输出的结果都是按找字典序来的，即1 2，1 3，2 1，2 3，3 1，3 2。这点是和交换法最不一样的地方。


- 代码的
```cpp
#include<iostream>
using namespace std;
#include<vector>
int arr[] = { 1,2,3,4 };
const int N = sizeof(arr) / sizeof(arr[0]);
bool state[N];//默认初始化为false，即0
vector<int>vec;
void func(int i)
{
	if (i == N)
	{
		for (int v : vec)
		{
			cout << v << " ";
		}
		cout << endl;
	}
	else
	{
		for (int k = 0; k < N; ++k)
		{
			if (!state[k])
			{
				state[k] = true;
				vec.push_back(arr[k]);
				func(i + 1);
				state[k] = false;
				vec.pop_back();
			}
		}
	}
}
int main()
{
	func(0);
	return 0;
}
```
