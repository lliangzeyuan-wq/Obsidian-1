- 树的所有叶子节点即为结果，叶子节点个数：n * n-1 * n-2 * …… * 2 * 1 一共为n!

```cpp
/*
解空间—排列数代码
*/
#include<iostream>
using namespace std;
void m_swap(int arr[], int i, int j)
{
	int temp = arr[i];
	arr[i] = arr[j];
	arr[j] = temp;
}
void func(int arr[],int i,int length)
{
	if (i == length)
	{
		for (int j = 0; j < length; ++j)
		{
			cout << arr[j] << " ";
		}
		cout << endl;
	}
	else
	{
		//生成i节点的所有孩子节点
		for (int k = i; k < length; ++k)
		{
			m_swap(arr, i, k);
			func(arr, i + 1, length);
			m_swap(arr,i, k);
		}
	}
}
int main()
{
	int arr[] = { 1,2,3,4 };
	int length = sizeof(arr) / sizeof(arr[0]);
	func(arr, 0, length);
	return 0;
}
```
