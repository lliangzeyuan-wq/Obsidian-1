```cpp
/*
挑选数字：有一组整数，请挑选出一组数字，让他们的和等于指定的值，存在解打印，不存在打印
*/
#include<iostream>
#include<vector>
using namespace std;
int number = 18;//指定的值
int arr[] = { 4,8,12,16,7,9,3 };
int length = sizeof(arr) / sizeof(arr[0]);
int sum = 0;//记录选择的数字的
vector<int>x;//记录选择的数字
int cnt = 0;//记录达到 i==length 的次数
int r = 0;//记录未处理的数字的和（不是未选择的！)
void func(int i)
{
	if (i == length)
	{
		cnt++;
		if (sum != number)
		{
			return;
		}
		for (int v : x)
		{
			cout << v << " ";
		}
		cout << endl;
	}
	else
	{
		r -= arr[i];//处理当前i节点
		if (sum + arr[i] <= number)//剪左树枝 已选择的数字的和+即将要选择的数字 < 设定的数字
		{
			x.push_back(arr[i]);
			sum += arr[i];
			func(i + 1);
			x.pop_back();
			sum -= arr[i];
		}

		if (sum + r >= number)//剪右树枝 已选择的数字的和 + 剩余的可以被选择的数字的和 > number
		{
			func(i + 1);
		}

		r += arr[i];//回溯的时候要记录成未处理的状态
	}
}
int main()
{
	for (int& v : arr)
	{
		r += v;
	}
	func(0);
	cout << cnt << endl;
	return 0;
}
```
