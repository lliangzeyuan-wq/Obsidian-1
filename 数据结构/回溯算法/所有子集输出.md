### 代码
```cpp
#include<iostream>
using namespace std;
void func(int arr[], int i, int length,int x[]) {
	if (i == length) {
		for (int j = 0; j < length; j++) {
			if (x[j] == 1) {
				cout << arr[j] << " ";
			}
		}
		cout << endl;
	}
	else {
		x[i] = 1;
		func(arr, i + 1, length, x);
		x[i] = 0;
		func(arr, i + 1, length, x);
	}
}
int main() {
	int arr[] = { 1,2,3 };
	int length = sizeof(arr) / sizeof(arr[0]);
	int x[3] = { 0 };
	func(arr,0,length,x);
	return 0;
}
```
