```
// 标准C++头文件，兼容C语言所有系统API
#include <iostream>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <pthread.h>

// 服务器监听端口（校园交友系统固定端口）
#define SERV_PORT 9527
// 最大支持的在线用户数量
#define MAX_USER 100

// 用户结构体：存储在线用户信息（C/C++通用）
typedef struct {
    char id[20];   // 学号
    int fd;        // 客户端socket
} User;

// 好友关系存储（内存版，重启消失）
typedef struct {
    char a[20];
    char b[20];
} FriendRelation;

FriendRelation friend_list[MAX_USER * 2];
int friend_count = 0;

//在线用户列表：储存所有连接到服务器的用户
User user_list[MAX_USER];
int user_count = 0;
// 互斥锁：保护共享数据（和C语言用法一样）
pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;

//线程处理函数：一个客户开一个线程
void* handle_client(void* arg);
//统一错误处理函数
void sys_err(const char* str);
// 查找用户fd
int find_user_fd(const char* id);
// 添加好友关系
void add_friend(const char* id1, const char* id2);

// 主函数：服务器入口
int main() {
    //文件描述符
    int lfd;
    //地址结构体
    struct sockaddr_in serv_addr;
    serv_addr.sin_family = AF_INET;//ipv4
    serv_addr.sin_port = htons(SERV_PORT);//host->net
    //服务端这里用INADDR_ANY,让服务器绑定本机所有ip，不管谁用哪个ip连，都能连接成功
    serv_addr.sin_addr.s_addr = htonl(INADDR_ANY);//host->net
    //客户端地址长度
    socklen_t clie_addr_len = sizeof(serv_addr);
    //线程id：用于创建新线程
    pthread_t tid;

    //socket
    lfd = socket(AF_INET, SOCK_STREAM, 0);
    if (lfd == -1) sys_err("socket创建失败");

    // 端口复用：解决端口占用问题
    int opt = 1;
    setsockopt(lfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    // 4. 绑定IP和端口到监听Socket 返回值：成功：0，失败：-1
    if (bind(lfd, (struct sockaddr*)&serv_addr, sizeof(serv_addr)) == -1)
        sys_err("bind绑定失败");

    // 3. 监听客户端（最大等待数列128）
    if (listen(lfd, 128) == -1) sys_err("listen监听失败");
    std::cout << "✅ 校园交友服务器启动成功！端口：" << SERV_PORT << "，等待用户连接..." << std::endl;

    // 4. 循环接收客户端（多线程）
    while (1) {
        int* cfd_ptr = new int;
        *cfd_ptr = accept(lfd, (struct sockaddr*)&serv_addr, &clie_addr_len);

        if (*cfd_ptr == -1) {
            perror("accept失败");
            delete cfd_ptr;
            continue;
        }

        std::cout << "🔗 新用户连接，socket=" << *cfd_ptr << std::endl;
        pthread_create(&tid, NULL, handle_client, (void*)cfd_ptr);
        pthread_detach(tid);
    }

    close(lfd);
    return 0;
}

// 查找用户是否在线
int find_user_fd(const char* id) {
    int fd = -1;
    pthread_mutex_lock(&mutex);
    for (int i = 0; i < user_count; i++) {
        if (strcmp(user_list[i].id, id) == 0) {
            fd = user_list[i].fd;
            break;
        }
    }
    pthread_mutex_unlock(&mutex);
    return fd;
}

// 添加好友关系
void add_friend(const char* id1, const char* id2) {
    pthread_mutex_lock(&mutex);
    strcpy(friend_list[friend_count].a, id1);
    strcpy(friend_list[friend_count].b, id2);
    friend_count++;
    pthread_mutex_unlock(&mutex);
}

// 线程函数：处理单个客户端
void* handle_client(void* arg) {
    int cfd = *(int*)arg;
    delete (int*)arg;

    char buf[1024];//数据缓存区：存放收发的信息
    int n;//接受数据的长度
    char user_id[20];//存储当前用户的学号

    // -------- 接收学号，登录 --------
    memset(buf, 0, sizeof(buf));
    n = read(cfd, buf, sizeof(buf));
    if (n <= 0) {
        close(cfd);
        return NULL;
    }
    strcpy(user_id, buf);
    std::cout << "🎉 用户 " << user_id << " 登录成功！socket=" << cfd << std::endl;

    // -------- 加入在线用户列表（加锁保护） --------
    pthread_mutex_lock(&mutex);
    strcpy(user_list[user_count].id, user_id);
    user_list[user_count].fd = cfd;
    user_count++;
    pthread_mutex_unlock(&mutex);

    // -------- 循环处理消息 --------
    while (1) {
        memset(buf, 0, sizeof(buf));
        n = read(cfd, buf, sizeof(buf));

        // 客户端断开连接
        if (n <= 0) {
            std::cout << "❌ 用户 " << user_id << " 断开连接，socket=" << cfd << std::endl;

            pthread_mutex_lock(&mutex);
            for (int i = 0; i < user_count; i++) {
                if (user_list[i].fd == cfd) {
                    for (int j = i; j < user_count - 1; j++)
                        user_list[j] = user_list[j + 1];
                    user_count--;
                    break;
                }
            }
            pthread_mutex_unlock(&mutex);

            close(cfd);
            break;
        }

        std::cout << "📩 收到用户 " << user_id << " 的消息：" << buf << std::endl;

        //-------------- 消息处理：聊天 + 好友邀请 -----------------------
        char target_id[20] = {0};
        char cmd[32] = {0};
        if (buf[0] == '@') {
            char* p = strchr(buf, ':');
            if (p != NULL) {
                *p = '\0';
                strcpy(target_id, buf + 1);
                strcpy(cmd, p + 1);

                // 1. 匹配邀请：@学号:MATCH
                if (strcmp(cmd, "MATCH") == 0) {
                    int target_fd = find_user_fd(target_id);
                    if (target_fd != -1) {
                        char invite_msg[256];
                        snprintf(invite_msg, sizeof(invite_msg),
                                 "📩 收到匹配邀请：用户 %s 想和你交友，回复 @%s:ACCEPT 同意 / @%s:REJECT 拒绝",
                                 user_id, user_id, user_id);
                        write(target_fd, invite_msg, strlen(invite_msg));
                        write(cfd, "✅ 邀请已发送，等待对方回复", strlen("✅ 邀请已发送，等待对方回复"));
                    } else {
                        write(cfd, "❌ 对方不在线，邀请发送失败", strlen("❌ 对方不在线，邀请发送失败"));
                    }
                }
                // 2. 同意好友：@学号:ACCEPT
                else if (strcmp(cmd, "ACCEPT") == 0) {
                    add_friend(user_id, target_id);
                    int inviter_fd = find_user_fd(target_id);
                    if (inviter_fd != -1) {
                        char success_msg[128] = "✅ 对方同意了你的邀请，你们已成为好友！";
                        write(inviter_fd, success_msg, strlen(success_msg));
                    }
                    write(cfd, "✅ 你已同意邀请，你们已成为好友！", strlen("✅ 你已同意邀请，你们已成为好友！"));
                }
                // 3. 拒绝好友：@学号:REJECT
                else if (strcmp(cmd, "REJECT") == 0) {
                    int inviter_fd = find_user_fd(target_id);
                    if (inviter_fd != -1) {
                        char reject_msg[128] = "❌ 对方拒绝了你的邀请";
                        write(inviter_fd, reject_msg, strlen(reject_msg));
                    }
                    write(cfd, "✅ 你已拒绝该邀请", strlen("✅ 你已拒绝该邀请"));
                }
                // 4. 普通聊天消息
                else {
                    char send_buf[1024];
                    snprintf(send_buf, sizeof(send_buf), "[%s]: %s", user_id, cmd);
                    int target_fd = find_user_fd(target_id);
                    if (target_fd != -1) {
                        write(target_fd, send_buf, strlen(send_buf));
                        std::cout << "✅ 转发给 " << target_id << " 成功！" << std::endl;
                    } else {
                        write(cfd, "❌ 对方不在线", strlen("❌ 对方不在线"));
                    }
                }
            } else {
                write(cfd, "❌ 格式错误：@学号:指令/消息", strlen("❌ 格式错误：@学号:指令/消息"));
            }
        } else {
            write(cfd, "❌ 格式错误，请使用：@学号:指令/消息内容", strlen("❌ 格式错误，请使用：@学号:指令/消息内容"));
        }
        //------------------ 消息处理结束 ----------------------------------
    }
    return NULL;
}

// 错误处理函数
void sys_err(const char* str) {
    perror(str);
    exit(1);
}
```

## 二、新版 `client.cpp`（优化交互提示）

```
#include <iostream>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

//服务器端口（必须和服务端完全一致）
#define SERV_PORT 9527

//错误处理函数
void sys_err(const char* str) {
    perror(str);
    exit(1);
}

int main() {
    int cfd;
    struct sockaddr_in serv_addr;
    serv_addr.sin_family = AF_INET;
    serv_addr.sin_port = htons(SERV_PORT);
    inet_pton(AF_INET, "127.0.0.1", &serv_addr.sin_addr);

    char buf[1024];
    int n;

    cfd = socket(AF_INET, SOCK_STREAM, 0);
    if (cfd == -1) sys_err("客户端socket创建失败");

    if (connect(cfd, (struct sockaddr*)&serv_addr, sizeof(serv_addr)) == -1)
        sys_err("连接服务器失败");

    // -------- 学号登录 --------
    std::cout << "请输入你的学号：";
    std::cin >> buf;
    write(cfd, buf, strlen(buf));
    std::cout << "✅ 学号 " << buf << " 登录成功！\n" << std::endl;
    std::cout << "💡 使用说明：" << std::endl;
    std::cout << "   - 聊天：@目标学号:消息内容" << std::endl;
    std::cout << "   - 邀请好友：@目标学号:MATCH" << std::endl;
    std::cout << "   - 同意好友：@邀请者学号:ACCEPT" << std::endl;
    std::cout << "   - 拒绝好友：@邀请者学号:REJECT" << std::endl;
    std::cout << "   - 退出：输入 quit\n" << std::endl;

    // -------- 消息收发 --------
    while (1) {
        memset(buf, 0, sizeof(buf));
        std::cout << "请输入指令/消息：";
        std::cin >> buf;

        if (strcmp(buf, "quit") == 0) {
            write(cfd, buf, strlen(buf));
            break;
        }

        write(cfd, buf, strlen(buf));

        memset(buf, 0, sizeof(buf));
        n = read(cfd, buf, sizeof(buf));
        if (n <= 0) break;

        std::cout << "📢 服务器通知：" << buf << "\n" << std::endl;
    }

    close(cfd);
    return 0;
}
```
