    // strchr 函数：在字符串中查找某个字符，找到返回地址，没找到返回NULL
    char* p = strchr(buf, ':');

    // 3. 如果找到了冒号，说明格式正确，开始解析
    if (p != NULL) {
        *p = '\0';  // 把冒号变成字符串结束符 \0，把原消息切成两段：@学号  和  消息内容

        // 4. 截取目标学号：buf+1 跳过@，剩下的就是学号
        strcpy(target_id, buf + 1);
        // 5. 截取消息内容：p+1 跳过冒号，剩下的就是消息
        strcpy(msg, p + 1);

        // 6. 拼接最终要发送给对方的消息格式：[发送者学号]：消息内容
        char send_buf[1024];

        int target_fd = -1;  // 目标用户的socket，默认-1表示不在线

1. 消息格式：`@学号:消息`
2. 找到 `:` 切成两段：**学号** + **消息**
3. 去用户列表**查学号是否在线**
4. 在线就**转发**，不在线就**返回错误**
