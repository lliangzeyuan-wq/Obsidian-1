    interests TEXT,                        -- 个人介绍

### 1. 新增依赖函数（MD5 加密 + Token 生成）

在 `server.cpp` 开头添加以下函数（需要链接 `openssl` 库，编译时加 `-lcrypto`）：

cpp

运行

```
#include <openssl/md5.h>   // MD5加密
#include <ctime>
#include <cstdlib>
#include <sstream>

// MD5密码加密
string md5(const string& str) {
    unsigned char digest[MD5_DIGEST_LENGTH];
    MD5((const unsigned char*)str.c_str(), str.size(), digest);
    char md5_str[33];
    for(int i = 0; i < MD5_DIGEST_LENGTH; i++)
        sprintf(&md5_str[i*2], "%02x", (unsigned int)digest[i]);
    return string(md5_str);
}

// 生成登录Token（时间戳+随机数，简单版）
string generateToken() {
    srand(time(0));
    stringstream ss;
    ss << time(0) << "_" << rand();
    return ss.str();
}

// Token验证：查询数据库是否有效，返回用户ID
string verifyToken(MYSQL* conn, const string& token) {
    if(token.empty()) return "";
    char sql[512];
    sprintf(sql, "SELECT id FROM users WHERE token='%s'", token.c_str());
    if(mysql_query(conn, sql)) return "";
    
    MYSQL_RES* res = mysql_store_result(conn);
    MYSQL_ROW row = mysql_fetch_row(res);
    string user_id = row ? row[0] : "";
    mysql_free_result(res);
    return user_id;
}
```

### 3. 改造「登录接口」`/api/login`  !!!

替换原登录接口逻辑，实现验证账号 + 生成 Token：

cpp

运行

```
// 登录接口
else if (method == "POST" && path == "/api/login") {
    try {
        json req = json::parse(body);
        string student_id = req["studentId"].get<string>();
        string password = req["password"].get<string>();
        string encrypted_pwd = md5(password);

        // 查询账号密码
        char sql[512];
        sprintf(sql, "SELECT id, name, gender, college, major, grade FROM users WHERE student_id='%s' AND password='%s'",
            student_id.c_str(), encrypted_pwd.c_str());

        if(mysql_query(conn, sql)) {
            resp_json["success"] = false;
            resp_json["message"] = "查询失败：" + string(mysql_error(conn));
        } else {
            MYSQL_RES* res = mysql_store_result(conn);
            MYSQL_ROW row = mysql_fetch_row(res);
            if(row) {
                // 生成Token并更新数据库
                string token = generateToken();
                char update_sql[512];
                sprintf(update_sql, "UPDATE users SET token='%s' WHERE student_id='%s'",
                    token.c_str(), student_id.c_str());
                mysql_query(conn, update_sql);

                resp_json["success"] = true;
                resp_json["data"] = {
                    {"id", row[0]},
                    {"studentId", student_id},
                    {"name", row[1]},
                    {"gender", row[2]},
                    {"college", row[3]},
                    {"major", row[4]},
                    {"grade", row[5]},
                    {"token", token},
                    {"profileCompleted", true}
                };
            } else {
                resp_json["success"] = false;
                resp_json["message"] = "学号或密码错误";
            }
            mysql_free_result(res);
        }
    } catch(...) {
        resp_json["success"] = false;
        resp_json["message"] = "参数错误";
    }
}
```

async request(url, options = {}) {
    if (API_CONFIG.useMock) return this.mockResponse(url, options);

    const fullUrl = API_CONFIG.baseUrl + url;
    const { method = 'GET', body, headers = {} } = options;
    const token = localStorage.getItem('token'); // 从本地取Token

    const fetchOptions = {
        method,
        headers: {
            'Content-Type': 'application/json',
            ...headers,
            // 自动携带Token
            ...(token ? { 'Authorization': `Bearer ${token}` } : {})
        }

    if (body && (method === 'POST' || method === 'PUT')) {

## ✅ 最终效果

1. **注册**：填写学号 / 姓名 / 密码 → 密码 MD5 加密存入 MySQL → 跳转到登录页
2. **登录**：验证学号 + 加密密码 → 生成 Token 存入数据库和前端 `localStorage` → 跳首页
3. **后续请求**：所有接口自动携带 Token → 后端验证后才允许访问（比如查看资料、发邀请）
4. **数据持久化**：账号 / 密码 / 个人信息永久存在 MySQL，告别模拟数据！
