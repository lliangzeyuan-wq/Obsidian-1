# 错误代码
### 问题
- 常规 SIGCHLD 信号不排队，15 个子进程几乎同时死亡，SIGCHLD 只触发 1 次回调；回调里`wait(NULL)`只回收 1 个，剩下 14 个全变僵尸 (defunct)
```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <signal.h>
#include <sys/wait.h>
#include <errno.h>
#include <pthread.h>

void sys_err(const char *str)
{
    perror(str);
    exit(1);
}

void catch_child(int signo)
{
    pid_t wpid;
    wpid = wait(NULL);
	if(wpid == -1)
	{
		sys_err("wait error");
	}
    printf("------------------------catch child id %d\n", wpid);
    return ;
}

int main(int argc, char *argv[])
{
    pid_t pid;
    int i;
    for (i = 0; i < 15; i++)
    {
        if ((pid = fork()) == 0)
            break;
    }

    if (15 == i)
    {
        struct sigaction act;
        act.sa_handler = catch_child;
        sigemptyset(&act.sa_mask);
        act.sa_flags = 0;

        sigaction(SIGCHLD, &act, NULL);
        printf("I'm parent, pid = %d\n", getpid());
		while(1);
    }
    else
    {
        printf("I'm child pid = %d\n", getpid());
    }

    return 0;
}
```



# 改进后的代码

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <signal.h>
#include <sys/wait.h>
#include <errno.h>
#include <pthread.h>

void sys_err(const char *str)
{
    perror(str);
    exit(1);
}

void catch_child(int signo)
{
    pid_t wpid;
	int status;
	//循环回收
	while( (wpid = waitpid(-1 , &status , 0 )) != -1  )
	{
		if(WIFEXITED(status))
			printf("------------------------catch child id %d, ret = %d\n", wpid , WEXITSTATUS(status));
	}
    return ;
}

int main(int argc, char *argv[])
{
	//阻塞SIG_BLOCK信号，等父进程注册之后再解开，防止子进程提前结束
	sigset_t mask;
	sigemptyset(&mask);
	sigaddset(&mask,SIGCHLD);
	sigprocmask(SIG_BLOCK , &mask , NULL);

    pid_t pid;
    int i;
    for (i = 0; i < 15; i++)
    {
        if ((pid = fork()) == 0)
            break;
    }

    if (15 == i)
    {
        struct sigaction act;
        act.sa_handler = catch_child;
        sigemptyset(&act.sa_mask);
        act.sa_flags = 0;

        sigaction(SIGCHLD, &act, NULL);
        printf("I'm parent, pid = %d\n", getpid());

		//解开阻塞
		sigprocmask(SIG_UNBLOCK, &mask ,NULL);

		while(1);
    }
    else
    {
        printf("I'm child pid = %d\n", getpid());
    }

    return 0;
}
```
