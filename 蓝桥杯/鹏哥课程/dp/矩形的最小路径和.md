#include <iostream>

using namespace std;

#include<cstring>

const int N=510;

int f[N][N];

int main()

{

    int n,m;

    cin>>n>>m;

    memset(f,0x3f,sizeof f);

    f[0][1]=0;

    for(int i=1;i<=n;++i)

    {

        for(int j=1;j<=m;++j)

        {

            int x;

            cin>>x;

            f[i][j]=min(f[i-1][j],f[i][j-1])+x;

        }

  

    }

  

    cout<<f[n][m]<<endl;

    return 0;

}
