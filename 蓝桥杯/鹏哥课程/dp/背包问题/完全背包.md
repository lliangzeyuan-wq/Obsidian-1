- 就状态转移方程这一个难点
- `f[i][j]=max(f[i-1][j],f[i][j-v[i]]+w[i])`
- 因此遍历的时候从左往右
![[Pasted image 20260323205927.png]]
- 其他的和01背包问题差不多
```cpp
#include <iostream>

using namespace std;

#include<cstring>

const int N = 1010;

int n, m;

int v[N], w[N]; //v:体积  w:价值

int f[N][N];

int main() {

    cin >> n >> m;

    for (int i = 1; i <= n; ++i) cin >> v[i] >> w[i];

    for (int i = 1; i <= n; ++i) {

        for (int j = 0; j <= m; ++j) {

            f[i][j] = f[i - 1][j];

            if (j >= v[i]) f[i][j] = max(f[i][j], f[i][j - v[i]] + w[i]);

        }

    }

    cout << f[n][m] << endl;

  

    memset(f, -0x3f, sizeof f);

    f[0][0] = 0;

    for (int i = 1; i <= n; ++i) {

        for (int j = 0; j <= m; ++j) {

            f[i][j] = f[i - 1][j];

            if (j >= v[i]) f[i][j] = max(f[i][j], f[i][j - v[i]] + w[i]);

        }

    }

    if(f[n][m]<0) cout<<0<<endl;

    else cout<<f[n][m]<<endl;

}
```
