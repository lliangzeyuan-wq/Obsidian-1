#include <iostream>

using namespace std;

const int N=1010;

int w[N],v[N];

int f[N][N];

int main()

{

    int n,m;//物品数量与背包容量。

    cin>>n>>m;

    for(int i=1;i<=n;++i)

    {

        cin>>w[i]>>v[i];

    }

    for(int i=1;i<=n;++i)

    {

        for(int j=0;j<=m;++j)

        {

            f[i][j]=f[i-1][j];

            if(j>=w[i])

            {

                f[i][j]=max(f[i][j],f[i-1][j-w[i]]+v[i]);

            }

        }

    }

    cout<<f[n][m]<<endl;

}
