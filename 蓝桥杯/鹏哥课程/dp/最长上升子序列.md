# 解法二

#include<iostream>
using namespace std;
const int N=5010;
int a[N];
int f[N],len;
int main()
{
    int n;
    cin>>n;
    for(int i=1;i<=n;++i)cin>>a[i];//o n
    for(int i=1;i<=n;++i)
    {
        if(len==0||a[i]>f[len]) f[++len]=a[i];
        else
        {
            //找到第一个>=x的位置
            //x=a[i]
            int l=1,r=len;
            while(l<r)
            {
                int mid=l+r>>1;
                if(f[mid]>=a[i]) r=mid;
                else l=mid+1;
            }
            f[l]=a[i];
        }
    }
    cout<<len<<endl;
    return 0;
}
