## 解法一
- 时间复杂度：O(n^2)，写法简单，适用于数据量小的题目
[4.4.1 【经典线性 dp 问题】最长上升子序列(一)_哔哩哔哩_bilibili](https://www.bilibili.com/video/BV1Wx2TBrEEp?spm_id_from=333.788.player.switch&vd_source=43c9de78f6e5f2b05790188e274ad943&p=156)

```cpp
#include<iostream>
using namespace std;
const int N=5010;
int a[N];
int f[N];
int main()
{
    int n;
    cin>>n;
    int ret=0;
    for(int i=1;i<=n;++i)cin>>a[i];
    for(int i=1;i<=n;++i)
    {
        f[i]=1;
        for(int j=1;j<i;++j)
        {
            if(a[j]<a[i])
            {
                f[i]=max(f[i],f[j]+1);
            }
        }
        ret=max(ret,f[i]);
    }
    cout<<ret<<endl;
    return 0;
}
```
# 解法二
- 时间复杂度：O(n* logn)，写法较复杂，适用于数据量小的题目
[4.4.2 【经典线性 dp 问题】最长上升子序列(二)_哔哩哔哩_bilibili](https://www.bilibili.com/video/BV1Wx2TBrEEp?spm_id_from=333.788.videopod.sections&vd_source=43c9de78f6e5f2b05790188e274ad943&p=157)

- f[i]：作为长度为i的顺序排列的最小的数字。eg:a[]={3,1,2};  则f[]={1,2}  f[1]=1而不为3为2腾出了位置，如果还是看不懂的化，建议结合视频，因为比较难讲清楚
```cpp
```cpp
#include<iostream>
using namespace std;
const int N=5010;
int a[N];
int f[N],len;
int main()
{
    int n;
    cin>>n;
    for(int i=1;i<=n;++i)cin>>a[i];//o n
    for(int i=1;i<=n;++i)
    {
        if(len==0||a[i]>f[len]) f[++len]=a[i];
        else
        {
            //找到第一个>=x的位置
            //x=a[i]
            int l=1,r=len;
            while(l<r)
            {
                int mid=l+r>>1;
                if(f[mid]>=a[i]) r=mid;
                else l=mid+1;
            }
            f[l]=a[i];
        }
    }
    cout<<len<<endl;
    return 0;
}
