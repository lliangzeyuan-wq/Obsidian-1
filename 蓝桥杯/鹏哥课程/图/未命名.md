# DFS
- ==一条路走到黑，走过的路不再走==
![[Pasted image 20260325202326.png]]
## 不带权值的dfs实现
```cpp
// 1. 引入C++的输入输出库，用来在屏幕上打印文字（cout就是用这个）
#include <iostream>
// 2. 引入动态数组vector，这是存储图的邻接表必须用的工具
#include <vector>
// 3. 使用标准命名空间，不用每次都写std::cout，直接写cout就行
using namespace std;

// 4. 定义最大顶点数
// 图里的顶点是 1、2、3、4、5、6 共6个点
// 数组下标从0开始，0号位置不用，所以定义N=7
const int N = 7;

// 5. 定义【邻接表】：图的核心存储结构
// vector<vector<int>> 是二维动态数组
// adj[u] 存储：从顶点 u 出发，能直接到达的所有顶点
vector<vector<int>> adj(N);

// 6. 定义【访问标记数组】
// 作用：记录哪个顶点已经遍历过了，避免重复走（防止死循环）
// 初始值全部是 false：代表所有点都没被访问过
vector<bool> visited(N, false);

// ==============================================
// 7. DFS 深度优先遍历函数
// 参数 x：代表【当前正在访问的顶点编号】
// 核心逻辑：一条路走到黑，走不通了再回头
// ==============================================
void dfs(int x) {
    // 第一步：标记当前顶点 x 为【已访问】
    // 走过的点，再也不走第二次
    visited[x] = true;

    // 第二步：打印当前访问的顶点，展示遍历顺序
    cout << x << " ";

    // 第三步：遍历 顶点x 的所有【邻居】（从x能直接到的所有点）
    // 循环语法：依次取出 adj[x] 里的每一个数，赋值给 y
    for (int y : adj[x]) {
        // 判断：如果邻居 y 还没有被访问过
        if (!visited[y]) {
            // 递归调用DFS：继续往深处走，访问顶点 y
            dfs(y);
        }
    }
    // 当x的所有邻居都遍历完了，函数自动返回（回溯）
}

// ==============================================
// 8. 主函数：程序的入口，所有代码从这里开始运行
// ==============================================
int main() {
    // ---------- 初始化邻接表（给图画地图）----------
    // 顶点1 可以直接到达：2、3
    adj[1] = {2, 3};
    // 顶点4 可以直接到达：5、6、1、3
    adj[4] = {5, 6, 1, 3};
    // 顶点5 可以直接到达：1
    adj[5] = {1};
    // 顶点6 可以直接到达：3
    adj[6] = {3};
    // 顶点2、3 没有出边，所以默认是空数组，不用写

    // 打印提示文字
    cout << "从顶点 1 开始的 DFS 遍历顺序：";

    // 调用DFS函数，从【顶点1】开始遍历整个图
    dfs(1);

    // 打印换行，让输出更整洁
    cout << endl;

    // 程序正常结束
    return 0;
}
```
## 带权值的dfs实现

#include<iostream>
using namespace std;
#include<vector>
const int N=7;
vector<vector<pair<int,int>>>adj(N);
vector<bool>visited(N,false);
void dfs(int x)
{
	visited[x]=true;
	cout<<x<<" ";
	for(auto &edge:adj[x])
	{
		int y=edge.first;
		if(!visited[y])
		{
			dfs(y);
		}
	}
}

	adj[1].emplace_back(2, 0);

    cout<<endl;

# BFS
- ==一层一层往外走，每个点只走一次==
- 和经典的用queue实现的
```cpp
// 1. 引入必要头文件
#include <iostream>
#include <vector>
#include <queue>  // BFS 核心：队列容器
using namespace std;

// 2. 定义常量与数据结构
const int N = 7;          // 顶点编号 1~6，数组开 7 个位置（下标 0 不用）
vector<vector<int>> adj(N); // 邻接表：adj[u] 存储顶点 u 的所有邻接点
vector<bool> vis(N, false); // 访问标记数组：vis[i]=true 表示顶点 i 已被遍历

// 3. BFS 核心函数：从顶点 start 开始遍历
void bfs(int start) {
    queue<int> q; // 待拓展的顶点队列（先进先出）

    // 初始化：起始点入队 + 标记为已访问
    q.push(start);
    vis[start] = true;

    // 循环：只要队列不为空，就继续遍历
    while (!q.empty()) {
        // 取出队首顶点 x（C++ 中 pop() 无返回值，需先 front() 再 pop()）
        int x = q.front();
        q.pop();

        // 输出当前访问的顶点（展示遍历顺序）
        cout << x << " ";

        // 遍历顶点 x 的所有邻接点 y
        for (int y : adj[x]) {
            // 如果邻接点 y 未被访问
            if (!vis[y]) {
                vis[y] = true;       // 标记为已访问（避免重复入队）
                q.push(y);           // 将 y 加入队列，等待后续处理
            }
        }
    }
}

// 4. 主函数：程序入口
int main() {
    // 初始化邻接表（和你提供的图完全一致）
    adj[1] = {2, 3};    // 顶点 1 → 2、3
    adj[3] = {4};        // 顶点 3 → 4
    adj[4] = {5, 6, 1, 3}; // 顶点 4 → 5、6、1、3
    adj[5] = {1};        // 顶点 5 → 1
    adj[6] = {3};        // 顶点 6 → 3
    // 顶点 2 无出边，adj[2] 为空

    // 执行 BFS 遍历（从顶点 1 开始）
    cout << "从顶点 1 开始的 BFS 遍历顺序：";
    bfs(1);
    cout << endl;

    return 0;
}
```


# 例题1
![[Pasted image 20260325213222.png]]
```cpp
// 包含C++标准输入输出库，用于控制台的输入输出
#include <iostream>
// 包含动态数组vector，用于存储图的邻接表和排序数据
#include <vector>
// 包含排序算法库，用于对帮派成员按规则排序
#include <algorithm>
// 包含栈容器，用于实现非递归DFS，避免递归栈溢出
#include <stack>
// 使用标准命名空间，简化代码书写（不用写std::）
using namespace std;

// 定义最大节点数，题目n≤1e5，所以开1e5+5的空间，防止越界
const int MAXN = 1e5 + 5;

// ===================== 全局变量定义 =====================
// 邻接表：存储树的结构
// adj[u] 存储所有与节点u直接相连的节点
vector<int> adj[MAXN];

// 子树大小数组
// size_[u] 表示：以u为根的子树中，总共有多少个节点（包含自己）
int size_[MAXN];

// 父节点数组
// parent[u] 表示：节点u的父节点是谁
// 作用：1. 找根节点 2. 遍历树时避免走回父节点（死循环）
int parent[MAXN];

// ===================== 非递归后序DFS函数 =====================
// 功能：计算每个节点的子树大小 size_[u]
// 优点：非递归写法，适配大数据（n=1e5），不会栈溢出
// 参数：root -> 树的根节点（帮派老大）
void dfs(int root) {
    // 定义栈，存储 pair<节点编号, 访问标记>
    // 访问标记：0=第一次访问（未处理），1=第二次访问（计算子树大小）
    stack<pair<int, int>> st;

    // 根节点入栈，标记为0（第一次访问）
    st.emplace(root, 0);
    // 根节点没有父节点，父节点设为0
    parent[root] = 0;

    // 循环遍历：栈不为空就继续执行
    while (!st.empty()) {
        // 取出栈顶元素：u=当前节点，visited=访问标记
        auto [u, visited] = st.top();
        // 弹出栈顶元素（必须弹出，否则栈不会变化）
        st.pop();

        // ========== 情况1：第一次访问节点u（标记=0） ==========
        if (!visited) {
            // 把u重新压回栈，标记改为1
            // 含义：等所有子节点遍历完，再回来计算u的子树大小
            st.emplace(u, 1);

            // 逆序遍历u的所有邻接节点
            // 栈是后进先出，逆序压栈才能保证正序遍历子节点
            for (auto it = adj[u].rbegin(); it != adj[u].rend(); ++it) {
                int v = *it;
                // 过滤父节点，只处理子节点
                if (v != parent[u]) {
                    parent[v] = u;       // 记录v的父节点是u
                    st.emplace(v, 0);    // 子节点v入栈，第一次访问
                }
            }
        }
        // ========== 情况2：第二次访问节点u（标记=1） ==========
        // 此时所有子节点已经遍历完成，可以计算子树大小
        else {
            // 子树大小初始为1（只算自己）
            size_[u] = 1;
            // 遍历所有邻接节点，累加子节点的子树大小
            for (int v : adj[u]) {
                if (v != parent[u]) {
                    // 子树大小 = 自己 + 所有子节点的子树总数
                    size_[u] += size_[v];
                }
            }
        }
    }
}

// ===================== 主函数：程序入口 =====================
int main() {
    // 加速输入输出：处理大数据时必须加，否则会超时
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    // n：帮派总人数  m：小明的编号
    int n, m;
    cin >> n >> m;

    // 初始化父节点数组，全部赋值为0
    fill(parent, parent + MAXN, 0);

    // 构建树：总共有n-1条边（树的边数=节点数-1）
    for (int i = 0; i < n - 1; ++i) {
        int l, r;
        cin >> l >> r;
        // l 隶属于 r → 无向图，双向加边
        adj[r].push_back(l);
        adj[l].push_back(r);
        // 标记l的父节点是r（确定父子关系）
        parent[l] = r;
    }

    // 寻找根节点（帮派老大）：父节点为0的节点就是根
    int root = 1;
    for (int i = 1; i <= n; ++i) {
        if (parent[i] == 0) {
            root = i;
            break;
        }
    }

    // 调用DFS，计算所有节点的子树大小
    dfs(root);

    // 收集数据，用于排序
    // pair格式：(-手下数, 成员编号)
    // 负号作用：让sort默认的升序，变成「手下数降序」
    vector<pair<int, int>> vec;
    for (int i = 1; i <= n; ++i) {
        // 手下数 = 子树大小 - 1（减去自己）
        vec.emplace_back(-(size_[i] - 1), i);
    }

    // 排序规则：
    // 1. 优先按 手下数 降序（负号实现）
    // 2. 手下数相同，按 编号 升序
    sort(vec.begin(), vec.end());

    // 查找小明的排名（排名从1开始）
    int rank = 0;
    for (int i = 0; i < vec.size(); ++i) {
        // 找到小明的编号m
        if (vec[i].second == m) {
            rank = i + 1;  // 数组下标从0开始，排名+1
            break;
        }
    }

    // 输出小明的最终排名
    cout << rank << endl;

    // 程序正常结束
    return 0;
}
```
