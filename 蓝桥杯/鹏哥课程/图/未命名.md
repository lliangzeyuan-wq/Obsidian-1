```cpp
// 1. 引入C++的输入输出库，用来在屏幕上打印文字（cout就是用这个）
#include <iostream>
// 2. 引入动态数组vector，这是存储图的邻接表必须用的工具
#include <vector>
// 3. 使用标准命名空间，不用每次都写std::cout，直接写cout就行
using namespace std;

// 4. 定义最大顶点数
// 图里的顶点是 1、2、3、4、5、6 共6个点
// 数组下标从0开始，0号位置不用，所以定义N=7
const int N = 7;

// 5. 定义【邻接表】：图的核心存储结构
// vector<vector<int>> 是二维动态数组
// adj[u] 存储：从顶点 u 出发，能直接到达的所有顶点
vector<vector<int>> adj(N);

// 6. 定义【访问标记数组】
// 作用：记录哪个顶点已经遍历过了，避免重复走（防止死循环）
// 初始值全部是 false：代表所有点都没被访问过
vector<bool> visited(N, false);

// ==============================================
// 7. DFS 深度优先遍历函数
// 参数 x：代表【当前正在访问的顶点编号】
// 核心逻辑：一条路走到黑，走不通了再回头
// ==============================================
void dfs(int x) {
    // 第一步：标记当前顶点 x 为【已访问】
    // 走过的点，再也不走第二次
    visited[x] = true;

    // 第二步：打印当前访问的顶点，展示遍历顺序
    cout << x << " ";

    // 第三步：遍历 顶点x 的所有【邻居】（从x能直接到的所有点）
    // 循环语法：依次取出 adj[x] 里的每一个数，赋值给 y
    for (int y : adj[x]) {
        // 判断：如果邻居 y 还没有被访问过
        if (!visited[y]) {
            // 递归调用DFS：继续往深处走，访问顶点 y
            dfs(y);
        }
    }
    // 当x的所有邻居都遍历完了，函数自动返回（回溯）
}

// ==============================================
// 8. 主函数：程序的入口，所有代码从这里开始运行
// ==============================================
int main() {
    // ---------- 初始化邻接表（给图画地图）----------
    // 顶点1 可以直接到达：2、3
    adj[1] = {2, 3};
    // 顶点4 可以直接到达：5、6、1、3
    adj[4] = {5, 6, 1, 3};
    // 顶点5 可以直接到达：1
    adj[5] = {1};
    // 顶点6 可以直接到达：3
    adj[6] = {3};
    // 顶点2、3 没有出边，所以默认是空数组，不用写

    // 打印提示文字
    cout << "从顶点 1 开始的 DFS 遍历顺序：";

    // 调用DFS函数，从【顶点1】开始遍历整个图
    dfs(1);

    // 打印换行，让输出更整洁
    cout << endl;

    // 程序正常结束
    return 0;
}
```
