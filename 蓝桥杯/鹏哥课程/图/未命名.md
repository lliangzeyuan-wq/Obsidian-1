# DFS
- ==一条路走到黑，走过的路不再走==

## 不带权值的dfs实现
```cpp
// 1. 引入C++的输入输出库，用来在屏幕上打印文字（cout就是用这个）
#include <iostream>
// 2. 引入动态数组vector，这是存储图的邻接表必须用的工具
#include <vector>
// 3. 使用标准命名空间，不用每次都写std::cout，直接写cout就行
using namespace std;

// 4. 定义最大顶点数
// 图里的顶点是 1、2、3、4、5、6 共6个点
// 数组下标从0开始，0号位置不用，所以定义N=7
const int N = 7;

// 5. 定义【邻接表】：图的核心存储结构
// vector<vector<int>> 是二维动态数组
// adj[u] 存储：从顶点 u 出发，能直接到达的所有顶点
vector<vector<int>> adj(N);

// 6. 定义【访问标记数组】
// 作用：记录哪个顶点已经遍历过了，避免重复走（防止死循环）
// 初始值全部是 false：代表所有点都没被访问过
vector<bool> visited(N, false);

// ==============================================
// 7. DFS 深度优先遍历函数
// 参数 x：代表【当前正在访问的顶点编号】
// 核心逻辑：一条路走到黑，走不通了再回头
// ==============================================
void dfs(int x) {
    // 第一步：标记当前顶点 x 为【已访问】
    // 走过的点，再也不走第二次
    visited[x] = true;

    // 第二步：打印当前访问的顶点，展示遍历顺序
    cout << x << " ";

    // 第三步：遍历 顶点x 的所有【邻居】（从x能直接到的所有点）
    // 循环语法：依次取出 adj[x] 里的每一个数，赋值给 y
    for (int y : adj[x]) {
        // 判断：如果邻居 y 还没有被访问过
        if (!visited[y]) {
            // 递归调用DFS：继续往深处走，访问顶点 y
            dfs(y);
        }
    }
    // 当x的所有邻居都遍历完了，函数自动返回（回溯）
}

// ==============================================
// 8. 主函数：程序的入口，所有代码从这里开始运行
// ==============================================
int main() {
    // ---------- 初始化邻接表（给图画地图）----------
    // 顶点1 可以直接到达：2、3
    adj[1] = {2, 3};
    // 顶点4 可以直接到达：5、6、1、3
    adj[4] = {5, 6, 1, 3};
    // 顶点5 可以直接到达：1
    adj[5] = {1};
    // 顶点6 可以直接到达：3
    adj[6] = {3};
    // 顶点2、3 没有出边，所以默认是空数组，不用写

    // 打印提示文字
    cout << "从顶点 1 开始的 DFS 遍历顺序：";

    // 调用DFS函数，从【顶点1】开始遍历整个图
    dfs(1);

    // 打印换行，让输出更整洁
    cout << endl;

    // 程序正常结束
    return 0;
}
```
## 带权值的dfs实现
```cpp
#include<iostream>
using namespace std;
#include<vector>
const int N=7;
vector<vector<pair<int,int>>>adj(N);
vector<bool>visited(N,false);
void dfs(int x)
{
	visited[x]=true;
	cout<<x<<" ";
	for(auto &edge:adj[x])
	{
		int y=edge.first;
		if(!visited[y])
		{
			dfs(y);
		}
	}
}
int main()
{
	adj[1].emplace_back(2, 0);  
    adj[1].emplace_back(3, 0);
    adj[4].emplace_back(5, 0);
    adj[4].emplace_back(6, 0);
    adj[3].emplace_back(4, 0);
    adj[5].emplace_back(1, 0);
    adj[6].emplace_back(3, 7);
    cout<<"从顶点 1 开始的 DFS 遍历顺序:"<<endl;
    dfs(1);
    cout<<endl;
	return 0;
}
```

# BFS
- ==一层一层往外走，每个点只走一次==
- 和经典的用queue实现的
```cpp
// 1. 引入必要头文件
#include <iostream>
#include <vector>
#include <queue>  // BFS 核心：队列容器
using namespace std;

// 2. 定义常量与数据结构
const int N = 7;          // 顶点编号 1~6，数组开 7 个位置（下标 0 不用）
vector<vector<int>> adj(N); // 邻接表：adj[u] 存储顶点 u 的所有邻接点
vector<bool> vis(N, false); // 访问标记数组：vis[i]=true 表示顶点 i 已被遍历

// 3. BFS 核心函数：从顶点 start 开始遍历
void bfs(int start) {
    queue<int> q; // 待拓展的顶点队列（先进先出）

    // 初始化：起始点入队 + 标记为已访问
    q.push(start);
    vis[start] = true;

    // 循环：只要队列不为空，就继续遍历
    while (!q.empty()) {
        // 取出队首顶点 x（C++ 中 pop() 无返回值，需先 front() 再 pop()）
        int x = q.front();
        q.pop();

        // 输出当前访问的顶点（展示遍历顺序）
        cout << x << " ";

        // 遍历顶点 x 的所有邻接点 y
        for (int y : adj[x]) {
            // 如果邻接点 y 未被访问
            if (!vis[y]) {
                vis[y] = true;       // 标记为已访问（避免重复入队）
                q.push(y);           // 将 y 加入队列，等待后续处理
            }
        }
    }
}

// 4. 主函数：程序入口
int main() {
    // 初始化邻接表（和你提供的图完全一致）
    adj[1] = {2, 3};    // 顶点 1 → 2、3
    adj[3] = {4};        // 顶点 3 → 4
    adj[4] = {5, 6, 1, 3}; // 顶点 4 → 5、6、1、3
    adj[5] = {1};        // 顶点 5 → 1
    adj[6] = {3};        // 顶点 6 → 3
    // 顶点 2 无出边，adj[2] 为空

    // 执行 BFS 遍历（从顶点 1 开始）
    cout << "从顶点 1 开始的 BFS 遍历顺序：";
    bfs(1);
    cout << endl;

    return 0;
}
```
