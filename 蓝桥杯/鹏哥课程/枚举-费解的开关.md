#include<bits/stdc++.h>
using namespace std;
const int N=5;
int a[N];
int t[N];

//用Brian Kernighan算法优化
int calc(int x)
{
	int cnt=0;
	while(x)
	{
		cnt++;
		x&=(x-1);
	}
	return cnt;   
}
int main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T;
	cin>>T;
	while(T--)
	{
		memset(a,0,sizeof(a));
		for(int i=0;i<N;++i)
		{
			for(int j=0;j<N;++j)
			{
				char c;
				cin>>c;
				if(c=='0')a[i]|=(1<<j);
			}
		}
		int ret=0x3f3f3f3f;
		
		//枚举第一行的所有按法
		for(int st=0;st<(1<<N);++st)
		{
			memcpy(t,a,sizeof(a));
			int push=st;
			int cnt=0;
			for(int i=0;i<N;++i)
			{
				cnt+=calc(push);
				t[i]=t[i]^push^(push>>1)^(push<<1);
				t[i]&=(1<<N)-1;
				t[i+1]^=push;
				push=t[i];
			}
			//他不能保证最后一行不是0
			if(a[N-1]==0) ret=min(ret,cnt);
		}
		
		if(ret>6) cout<<-1<<endl;
		else cout<<ret<<endl;
	}
	return 0;
}
