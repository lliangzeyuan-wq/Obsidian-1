```cpp
#include<bits/stdc++.h>
using namespace std;
bool getColor(int n,int k)
{
	int cnt=0;
	int pos=k;
	for(int i=n;i>1;--i)
	{
		if(pos%2==0)
		{
			cnt++;
		}
		pos=(pos+1)/2;
	}
	
	return (cnt%2==0);
}
int main()
{
	ios::sync_with_stdio(false);
	cin.tie(NULL);
	
	int m;
	cin>>m;
	while(m--)
	{
		int n,k;
		cin>>n>>k;
		if(getColor(n,k))
		{
			cout<<"RED"<<endl;
		}
		else
		{
			cout<<"BLACK"<<endl;
		}
	}
	
	return 0;
}
```
