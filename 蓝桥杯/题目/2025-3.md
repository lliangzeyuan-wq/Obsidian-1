### 蓝桥杯 2025 省赛真题 - 黑客（带详细注释版）

```cpp
#include<bits/stdc++.h>
// 宏定义，将 long long 简写为 ll，减少代码长度
#define ll long long
using namespace std;

// 全局数组定义（全局数组默认初始化为0，且内存空间更大）
const int MAX = 500001;
int counts[MAX]; // 计数数组：counts[x] 表示数字x出现的次数
ll inv_fac[MAX]; // 逆元阶乘数组：inv_fac[i] = (i!)^{-1} mod mod
ll fac[MAX];     // 阶乘数组：fac[i] = i! mod mod


// 快速幂函数：计算 (a^b) % mod，用于求逆元
// a: 底数, b: 指数, mod: 模数
ll pow(ll a, int b, int mod) {
    ll res = 1; // 结果初始化为1
    while (b) { // 快速幂核心循环，当指数不为0时
        if (b & 1) // 如果指数的二进制最后一位是1
            res = res * a % mod; // 结果乘上当前底数，并取模
        a = a * a % mod; // 底数平方，取模
        b >>= 1; // 指数右移一位（等价于除以2）
    }
    return res;
}

int main() {
    // 加速输入输出（蓝桥杯大数据量必备，防止超时）
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    const int mod = 1e9 + 7; // 题目要求的模数
    int n; // 输入的总元素个数（即题目中的 k = n*m + 2）
    cin >> n;
    
    int nums[n]; // 存储输入的所有数字
    for (int i = 0; i < n; ++i) {
        int x;
        cin >> x;
        counts[x]++; // 统计每个数字的出现次数
        nums[i] = x; // 暂存数字（原代码未用到，保留以兼容原图）
    }

    // 核心转换：s = n*m = 总个数 - 2（减去前两个数n和m）
    // 这里直接复用n变量，此时n代表矩阵元素的总个数 s
    n -= 2;

    // 1. 预处理阶乘数组
    fac[0] = 1; // 0! = 1
    for (int i = 1; i <= 500000; ++i) {
        fac[i] = fac[i - 1] * i % mod;
    }

    // 2. 预处理逆元阶乘数组（费马小定理）
    // 先求最大数的逆元，再倒推
    inv_fac[500000] = pow(fac[500000], mod - 2, mod);
    for (int i = 499999; i >= 0; --i) {
        inv_fac[i] = inv_fac[i + 1] * (i + 1) % mod;
    }

    // 3. 计算基础排列数 a = (n)! / (c1! * c2! * ... * ck!) mod mod
    // 即：矩阵元素的不重复全排列数
    ll a = fac[n]; 
    // 【修正点】原代码 for(int x:counts) 会遍历50万次且包含大量0，效率低且逻辑有瑕疵
    // 正确做法是遍历所有出现过的数字，这里用set去重后遍历nums数组
    unordered_set<int> st(nums, nums + (n + 2)); // 放入所有出现过的数字
    for (int x : st) {
        if (counts[x] >= 1) { // 只有出现过的数字才计算
            a = a * inv_fac[counts[x]] % mod;
        }
    }

    ll ans = 0; // 最终答案，初始化为0

    // 4. 枚举所有可能的因数对 (i, j)，满足 i * j = n (即s)
    // 枚举到 sqrt(n) 以优化时间复杂度
    for (int i = 1; i * i <= n; ++i) {
        // 剪枝1：i不是n的因数，或数字i根本没出现过，直接跳过
        if (n % i != 0 || counts[i] == 0)
            continue;
        
        int j = n / i; // 对应的另一个因数
        
        // 剪枝2：数字j没出现过，或者 i==j但j的出现次数不足2次，跳过
        if (counts[j] == 0 || (i == j && counts[j] < 2))
            continue;

        // 情况一：行数 == 列数 (i == j)
        if (i == j) {
            // 选法数：C(counts[i], 2) = counts[i]! / [(counts[i]-2)! * 2!]
            // 总贡献：基础排列数a * 选法数
            ll c = fac[counts[i]] * inv_fac[counts[i] - 2] % mod;
            ans = (ans + a * c)%mod;
        }
        // 情况二：行数 != 列数 (i != j)
        else {
            // 选法数：counts[i] * counts[j] (选一个i，选一个j)
            // 因为(i,j)和(j,i)是两种不同矩阵，所以最后要 *2
            ll c1 = fac[counts[i]] * inv_fac[counts[i] - 1] % mod; // 等价于 counts[i]
            ll c2 = fac[counts[j]] * inv_fac[counts[j] - 1] % mod; // 等价于 counts[j]
            ll c = c1 * c2 % mod * 2 % mod; // 乘2处理两种顺序
            ans = (ans + a * c) % mod;
        }
    }

    // 输出最终答案
    cout << ans << endl;
    return 0;
}
```
