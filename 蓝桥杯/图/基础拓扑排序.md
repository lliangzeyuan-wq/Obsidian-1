```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
    // 1. 输入节点数 n 和边数 m
    int n, m;
    cin >> n >> m;

    // 2. 邻接表存储图（1-based 编号，和原代码一致）
    vector<vector<int>> adj(n + 1);
    // 入度数组：记录每个节点的入度（有多少个前置依赖）
    vector<int> in_degree(n + 1, 0);

    // 3. 读入 m 条边，构建邻接表和入度
    for (int i = 0; i < m; ++i) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y);    // 边 x → y
        in_degree[y]++;         // y 的入度 +1
    }

    // 4. 初始化队列：将所有「入度为 0」的节点入队（没有依赖的节点）
    queue<int> q;
    for (int i = 1; i <= n; ++i) {
        if (in_degree[i] == 0) {
            q.push(i);
        }
    }

    // 5. BFS 执行拓扑排序
    vector<int> topo_order; // 保存最终的拓扑序列
    while (!q.empty()) {
        int x = q.front();
        q.pop();
        topo_order.push_back(x); // 将当前节点加入拓扑序列

        // 遍历 x 的所有邻接点 y（x → y）
        for (int y : adj[x]) {
            in_degree[y]--; // 处理完 x，y 的入度减 1（依赖减少一个）
            // 如果 y 的入度变为 0，说明所有前置依赖都处理完了，可以入队
            if (in_degree[y] == 0) {
                q.push(y);
            }
        }
    }

    // 6. 输出结果
    cout << "拓扑序列：";
    for (int num : topo_order) {
        cout << num << " ";
    }
    cout << endl;

    // 7. 检测是否有环（拓扑序列长度 != 节点数 → 存在环）
    if (topo_order.size() != n) {
        cout << "⚠️ 图中存在环，无法完成拓扑排序！" << endl;
    }

    return 0;
}
```

# 解决DAG问题
```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <algorithm> // 用于max函数
using namespace std;

int main() {
    int n, m; // n:节点数  m:边数
    cin >> n >> m;

    vector<vector<int>> adj(n + 1); // 邻接表存储图
    vector<int> in_degree(n + 1, 0); // 入度数组
    vector<int> dp(n + 1, 1); // DP数组：dp[i]表示以i结尾的最长路径长度（初始为1，节点自身）

    // 1. 读入边，构建邻接表+统计入度
    for (int i = 0; i < m; ++i) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y); // 有向边 x→y
        in_degree[y]++;       // y的入度+1
    }

    queue<int> q;
    // 2. 初始化队列：入度为0的节点（无依赖）入队
    for (int i = 1; i <= n; ++i) {
        if (in_degree[i] == 0) {
            q.push(i);
        }
    }

    // 3. 拓扑排序 + DP状态转移（核心部分）
    while (!q.empty()) {
        int x = q.front();
        q.pop();

        // 遍历x的所有邻接点y（x→y）
        for (int y : adj[x]) {
            // ✅ DP状态转移：以y结尾的最长路径 = max(自身, 以x结尾的最长路径+1)
            dp[y] = max(dp[y], dp[x] + 1);

            // 处理入度：x处理完后，y的入度-1
            in_degree[y]--;
            // 如果y的入度变为0，说明所有前置依赖已处理完，可以入队
            if (in_degree[y] == 0) {
                q.push(y);
            }
        }
    }

    // 4. 输出结果（以最长路径为例）
    int max_path = 0;
    for (int i = 1; i <= n; ++i) {
        max_path = max(max_path, dp[i]);
    }
    cout << "DAG最长路径长度: " << max_path << endl;

    // 可选：输出每个节点的DP值（查看路径细节）
    // for (int i = 1; i <= n; ++i) {
    //     cout << "dp[" << i << "] = " << dp[i] << endl;
    // }

    return 0;
}
```
