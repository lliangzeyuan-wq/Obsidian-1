```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
    // 1. 输入节点数 n 和边数 m
    int n, m;
    cin >> n >> m;

    // 2. 邻接表存储图（1-based 编号，和原代码一致）
    vector<vector<int>> adj(n + 1);
    // 入度数组：记录每个节点的入度（有多少个前置依赖）
    vector<int> in_degree(n + 1, 0);

    // 3. 读入 m 条边，构建邻接表和入度
    for (int i = 0; i < m; ++i) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y);    // 边 x → y
        in_degree[y]++;         // y 的入度 +1
    }

    // 4. 初始化队列：将所有「入度为 0」的节点入队（没有依赖的节点）
    queue<int> q;
    for (int i = 1; i <= n; ++i) {
        if (in_degree[i] == 0) {
            q.push(i);
        }
    }

    // 5. BFS 执行拓扑排序
    vector<int> topo_order; // 保存最终的拓扑序列
    while (!q.empty()) {
        int x = q.front();
        q.pop();
        topo_order.push_back(x); // 将当前节点加入拓扑序列

        // 遍历 x 的所有邻接点 y（x → y）
        for (int y : adj[x]) {
            in_degree[y]--; // 处理完 x，y 的入度减 1（依赖减少一个）
            // 如果 y 的入度变为 0，说明所有前置依赖都处理完了，可以入队
            if (in_degree[y] == 0) {
                q.push(y);
            }
        }
    }

    // 6. 输出结果
    cout << "拓扑序列：";
    for (int num : topo_order) {
        cout << num << " ";
    }
    cout << endl;

    // 7. 检测是否有环（拓扑序列长度 != 节点数 → 存在环）
    if (topo_order.size() != n) {
        cout << "⚠️ 图中存在环，无法完成拓扑排序！" << endl;
    }

    return 0;
}
```
