#include <vector>
#include <algorithm>
using namespace std;

// 并查集类（Union-Find / DSU），完全对应原Java UF类
class UF {
private:
    int count;          // 连通分量个数
    vector<int> parent; // 父节点数组
    vector<int> size;   // 树的大小（重量，用于按秩合并）

    // 私有方法：查找节点x的根节点，带隔代路径压缩（和原Java逻辑完全一致）
    int find(int x) {
        while (parent[x] != x) {
            // 路径压缩：将x直接指向爷爷节点，扁平化树结构
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }

public:
    // 构造函数：初始化n个节点（0 ~ n-1）
    UF(int n) {
        count = n;
        parent.resize(n);
        size.resize(n, 1); // 初始每个节点单独成树，大小为1
        for (int i = 0; i < n; ++i) {
            parent[i] = i; // 初始父节点为自身（根节点）
        }
    }

    // 合并节点p和q所在的连通分量（按size合并，保持树平衡）
    // ⚠️ 关键修改：C++中union是关键字，因此方法名改为unite
    void unite(int p, int q) {
        int rootP = find(p);
        int rootQ = find(q);
        if (rootP == rootQ) {
            return; // 已经连通，无需合并
        }

        // 小树挂到大树上，避免树高过高影响效率（和原Java逻辑完全一致）
        if (size[rootP] > size[rootQ]) {
            parent[rootQ] = rootP;
            size[rootP] += size[rootQ];
        } else {
            parent[rootP] = rootQ;
            size[rootQ] += size[rootP];
        }

        count--; // 连通分量数减1
    }

    // 判断节点p和q是否连通（原Java connected方法）
    bool connected(int p, int q) {
        return find(p) == find(q);
    }

    // 获取当前连通分量的总个数（原Java count()方法）
    // ⚠️ 优化：避免和成员变量count冲突，改名为getCount
    int getCount() {
        return count;
    }
};

// 力扣题目Solution类
class Solution {
public:
    int minimumCost(int n, vector<vector<int>>& connections) {
        // 城市编号为1~n，因此初始化大小为n+1（0号节点闲置，和原Java逻辑一致）
        UF uf(n + 1);

        // 对所有边按权重从小到大排序（Kruskal算法核心步骤）
        // 对应Java: Arrays.sort(connections, (a,b) -> a[2]-b[2])
        sort(connections.begin(), connections.end(), 
            [](sslocal://flow/file_open?url=const+vector%3Cint%3E%26+a%2C+const+vector%3Cint%3E%26+b&flow_extra=eyJsaW5rX3R5cGUiOiJjb2RlX2ludGVycHJldGVyIn0=) {
                return a[2] < b[2]; // 升序排序，权重小的优先
            });

        int mst = 0; // 最小生成树的总权重
        for (auto& edge : connections) {
            int u = edge[0];
            int v = edge[1];
            int weight = edge[2];

            // 如果两节点已连通，说明这条边会形成环，跳过（和原Java逻辑一致）
            if (uf.connected(u, v)) {
                continue;
            }

            // 否则，将这条边加入最小生成树，合并连通分量
            mst += weight;
            uf.unite(u, v);
        }

        // 由于0号节点闲置，占用1个连通分量
        // 因此当总连通分量数为2时，说明1~n所有城市完全连通（和原Java逻辑一致）
        return uf.getCount() == 2 ? mst : -1;
    }
};
