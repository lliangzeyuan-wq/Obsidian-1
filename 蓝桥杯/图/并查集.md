```cpp
#include <vector>
using namespace std;

// 并查集（Union-Find / DSU）最终版
// 核心优化：按大小(size)合并 + 路径压缩，时间复杂度近乎O(1)
class UF {
private:
    // ================================== 私有成员变量 ==================================
    int count;          // 记录当前图中的【连通分量总数】（有多少个独立的连通块）
    vector<int> parent; // 父节点数组：parent[x] 表示节点 x 的父节点
                        // 根节点的唯一特征：parent[x] == x（自己指向自己）
    vector<int> size;   // 重量数组：size[root] 表示以 root 为根的树的总节点数
                        // 【仅根节点的size有效】，用于「按大小合并」，保证树的平衡，避免退化成链表


    // ================================== 私有核心方法：find（带路径压缩） ==================================
    // 功能：查找节点 x 的根节点，同时进行【路径压缩】，扁平化树结构
    // 路径压缩：在找根的过程中，将 x 直接指向爷爷节点，大幅降低树高，让后续操作更快
    int find(int x) {
        // 循环向上找根，直到 x 是根节点（parent[x] == x）
        while (parent[x] != x) {
            // 【核心路径压缩代码】：将 x 的父节点更新为 x 的「爷爷节点」（父节点的父节点）
            // 相当于让 x 向上跳一步，直接缩短到根的路径，把树压扁
            parent[x] = parent[parent[x]];
            // x 移动到新的父节点（原爷爷节点），继续向上找根
            x = parent[x];
        }
        // 返回 x 的根节点
        return x;
    }


public:
    // ================================== 公有构造函数 ==================================
    // 功能：初始化并查集，n 为图的节点总数（默认 0-based 编号：0 ~ n-1）
    UF(int n) {
        // 初始状态：所有节点互不连通，连通分量数 = 节点总数 n
        this->count = n;
        // 初始化父节点数组，长度为 n，对应 n 个节点
        parent.resize(n);
        // 初始化重量数组，长度为 n，初始每个节点的树大小为 1
        size.resize(n);

        // 遍历每个节点，初始化父节点和重量
        for (int i = 0; i < n; ++i) {
            parent[i] = i;  // 初始时，每个节点的父节点是自己（自己是根节点）
            size[i] = 1;    // 初始时，每个节点单独成树，树的大小（重量）为 1
        }
    }


    // ================================== 公有方法：unite（原 union，C++ 关键字冲突改名） ==================================
    // 功能：合并节点 p 和 q 所在的两个连通分量（按大小合并，保证树平衡）
    // 注意：C++ 中 union 是关键字，无法用作函数名，因此改名为 unite，逻辑完全不变
    void unite(int p, int q) {
        // 1. 分别找到 p 和 q 的根节点（find 过程会自动进行路径压缩）
        int rootP = find(p);
        int rootQ = find(q);

        // 2. 如果 p 和 q 已经在同一个连通分量（根节点相同），无需合并，直接返回
        if (rootP == rootQ) {
            return;
        }

        // 3. 【按大小合并核心逻辑】：比较两棵树的重量，永远将「小树」挂到「大树」的根下
        // 这样可以保证树的高度始终维持在极低水平，彻底避免树退化成链表
        if (size[rootP] > size[rootQ]) {
            // 情况1：rootP 所在的树更大 → 将 rootQ（小树）的父节点设为 rootP（大树的根）
            parent[rootQ] = rootP;
            // 更新大树的重量：合并后总节点数 = 原大树节点数 + 原小树节点数
            size[rootP] += size[rootQ];
        } else {
            // 情况2：rootQ 所在的树更大（或重量相等） → 将 rootP（小树）的父节点设为 rootQ（大树的根）
            parent[rootP] = rootQ;
            // 更新大树的重量：合并后总节点数 = 原大树节点数 + 原小树节点数
            size[rootQ] += size[rootP];
        }

        // 4. 两个连通分量合并为一个，连通分量总数减 1
        count--;
    }


    // ================================== 公有方法：connected ==================================
    // 功能：判断节点 p 和 q 是否连通（是否在同一个连通分量中）
    bool connected(int p, int q) {
        // 两个节点连通的充要条件：它们的根节点完全相同
        // find 过程会自动进行路径压缩，完全不影响连通性判断的正确性
        int rootP = find(p);
        int rootQ = find(q);
        return rootP == rootQ;
    }


    // ================================== 公有方法：count ==================================
    // 功能：返回当前图中的连通分量总数
    int count() {
        return count;
    }
};


// ================================== 可选：测试用例（验证逻辑正确性） ==================================
#include <iostream>
int main() {
    // 初始化 10 个节点（0~9），对应你之前的例子
    UF uf(10);
    cout << "初始连通分量数：" << uf.count() << endl; // 输出：10

    // 合并 0 和 1
    uf.unite(0, 1);
    cout << "合并0-1后，连通分量数：" << uf.count() << endl; // 输出：9
    cout << "0和1是否连通：" << boolalpha << uf.connected(0, 1) << endl; // 输出：true

    // 合并 1 和 2
    uf.unite(1, 2);
    cout << "合并1-2后，连通分量数：" << uf.count() << endl; // 输出：8
    cout << "0和2是否连通：" << boolalpha << uf.connected(0, 2) << endl; // 输出：true

    return 0;
}
```
