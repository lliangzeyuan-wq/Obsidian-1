# Dijkstra算法

```cpp
// 包含输入输出流，用于 cin/cout
#include <iostream>
// 包含动态数组 vector，用来存图
#include <vector>
// 包含优先队列，实现堆优化 Dijkstra
#include <queue>
// 使用标准命名空间，省去每次写 std::
using namespace std;

// 给 long long 起别名 ll，写代码更短，避免溢出
typedef long long ll;
// 定义无穷大：比题目最大可能距离还大，代表初始不可达
const ll INF = 1e18;

int main() {
    // ===================== 加速 cin/cout，大数据必加 =====================
    // 关闭 cin 和 C 语言 stdio 的同步，提速
    ios::sync_with_stdio(false);
    // 解除 cin 和 cout 的绑定，进一步提速
    cin.tie(nullptr);

    // N = 建筑数量（节点数），M = 单向道路数量（边数）
    int N, M;
    cin >> N >> M;

    // ===================== 用邻接表存图 =====================
    // adj[u] 存储所有从 u 出发的边：pair(目标点v, 边权w)
    vector<vector<pair<int, ll>>> adj(N + 1);

    // 循环读入 M 条边
    for (int i = 0; i < M; ++i) {
        int u, v;   // u 起点，v 终点
        ll w;       // 边的长度/权重
        cin >> u >> v >> w;
        // 把这条 u→v 的边加入邻接表
        adj[u].push_back(make_pair(v, w));
    }

    // ===================== 初始化最短距离数组 =====================
    // dist[i] = 从 1 号点到 i 号点的最短距离
    // 一开始全部设为无穷大，表示还没走到
    vector<ll> dist(N + 1, INF);
    // 起点是 1 号，自己到自己距离为 0
    dist[1] = 0;

    // ===================== 小根堆（优先队列） =====================
    // 优先队列里存 pair(当前最短距离, 节点编号)
    // greater<> 表示从小到大排，每次取距离最小的点
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;
    // 把起点（距离0，编号1）放进堆里
    pq.push(make_pair(0, 1));

    // ===================== Dijkstra 核心循环 =====================
    // 堆不为空就一直循环
    while (!pq.empty()) {
        // 取出堆顶：当前距离最小的点
        pair<ll, int> curr = pq.top();
        pq.pop();  // 弹出堆顶

        ll d = curr.first;   // d = 这个点当前记录的距离
        int u = curr.second;  // u = 这个点的编号

        // ===================== 重要优化 =====================
        // 如果这个距离 d 已经比我们算好的最短距离还大
        // 说明这条是旧数据，没用了，直接跳过
        if (d > dist[u]) continue;

        // ===================== 遍历 u 的所有邻边 =====================
        for (int i = 0; i < adj[u].size(); ++i) {
            int v = adj[u][i].first;   // 这条边指向的点 v
            ll w = adj[u][i].second;  // 这条边的长度 w

            // ===================== 松弛操作（核心） =====================
            // 如果 1→u→v 比 1→v 目前的距离更短
            if (dist[v] > dist[u] + w) {
                // 更新 v 的最短距离
                dist[v] = dist[u] + w;
                // 把新距离和 v 点丢进堆，后续继续更新
                pq.push(make_pair(dist[v], v));
            }
        }
    }

    // ===================== 输出答案 =====================
    for (int i = 1; i <= N; ++i) {
        // 除了第一个数，前面都加空格，格式正确
        if (i > 1) cout << ' ';
        // 输出 1 到 i 的最短距离
        cout << dist[i];
    }
    cout << endl;

    return 0;
}
```
