#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

// ==================== 并查集（Union-Find / DSU）类 ====================
// 作用：快速判断连通性、合并连通分量，是Kruskal算法的核心工具
class UF {
private:
    int count;          // 连通分量的总个数（初始为城市数N，对应1~N的节点）
    vector<int> parent; // parent[i] 表示节点i的父节点，用于找根节点
    vector<int> size;   // size[i] 表示以i为根的树的节点数，用于「按秩合并」（保持树平衡，提升效率）

    // 私有方法：查找节点x的根节点，带【路径压缩】（隔代压缩，循环实现，无栈溢出风险）
    int find(int x) {
        // 循环向上找根，直到x的父节点是自己（根节点的特征：parent[x] == x）
        while (parent[x] != x) {
            // 🔹 路径压缩核心：将x的父节点直接指向「爷爷节点」，扁平化树结构
            // 作用：大幅降低树高，让后续find操作时间复杂度接近O(1)
            parent[x] = parent[parent[x]];
            // x移动到新的父节点（原爷爷节点），继续向上找根
            x = parent[x];
        }
        return x;
    }

public:
    // 🔹 构造函数：初始化1~n的节点（1-based，完美对应题目中的城市编号）
    UF(int n) {
        count = n;                  // 初始状态：每个城市是独立的连通分量，共n个
        parent.resize(n + 1);       // 节点编号1~n，数组大小设为n+1，0号闲置不用
        size.resize(n + 1, 1);      // 初始每个节点单独成树，树的大小为1
        // 遍历1~n的所有节点，初始化父节点和树大小
        for (int i = 1; i <= n; ++i) {
            parent[i] = i;  // 初始时，每个节点的父节点是自己（自己是根节点）
            size[i] = 1;    // 初始时，每个节点的树大小为1
        }
    }

    // 🔹 合并节点p和q所在的连通分量（按树大小合并，小树挂大树，保持树平衡）
    void unite(int p, int q) {
        int rootP = find(p); // 找到p的根节点
        int rootQ = find(q); // 找到q的根节点
        if (rootP == rootQ) {
            return; // p和q已经在同一个连通分量，无需合并（避免成环）
        }

        // 按树大小合并：将小树挂到大树的根节点下，避免树过高，保证效率
        if (size[rootP] > size[rootQ]) {
            parent[rootQ] = rootP;  // 小树rootQ的父节点设为大树rootP
            size[rootP] += size[rootQ]; // 大树的大小累加小树的大小
        } else {
            parent[rootP] = rootQ;  // 小树rootP的父节点设为大树rootQ
            size[rootQ] += size[rootP]; // 大树的大小累加小树的大小
        }

        count--; // 两个连通分量合并为一个，连通分量数-1
    }

    // 🔹 判断节点p和q是否连通（是否在同一个连通分量）
    bool connected(int p, int q) {
        return find(p) == find(q); // 根节点相同 → 连通
    }

    // 🔹 获取当前连通分量的总个数
    int getCount() {
        return count;
    }
};

// ==================== 主函数（Kruskal算法实现） ====================
int main() {
    // 🔹 输入加速优化：蓝桥OJ大数据量必加，避免cin超时
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T; // 测试用例的数量
    cin >> T;
    while (T--) { // 循环处理每个测试用例
        int N, M; // N：城市数量，M：道路数量
        cin >> N >> M;

        // 🔹 存储所有道路（边），每个边的格式：{城市X, 城市Y, 道路长度C}
        vector<vector<int>> edges(M);
        for (int i = 0; i < M; ++i) {
            int X, Y, C;
            cin >> X >> Y >> C;
            edges[i] = {X, Y, C};
        }

        // 🔹 Kruskal核心步骤1：按道路长度（权值）从小到大排序所有边
        // 贪心思想：优先选最短的边，保证最终生成树的最大边尽可能小
        sort(edges.begin(), edges.end(), [](sslocal://flow/file_open?url=const+vector%3Cint%3E%26+a%2C+const+vector%3Cint%3E%26+b&flow_extra=eyJsaW5rX3R5cGUiOiJjb2RlX2ludGVycHJldGVyIn0=) {
            return a[2] < b[2]; // 按边的第3个元素（长度C）升序排序
        });

        // 🔹 初始化并查集，处理1~N的城市节点
        UF uf(N);
        int max_fuel = 0; // 记录最小生成树中的最大边长度 → 就是所需的最小油箱容量

        // 🔹 Kruskal核心步骤2：遍历排序后的边，合并连通分量，同时记录最大边
        for (const auto& edge : edges) {
            int X = edge[0]; // 边的起点城市X
            int Y = edge[1]; // 边的终点城市Y
            int C = edge[2]; // 边的长度C（对应耗油量，每公里1升）

            // 如果X和Y已经连通 → 这条边会形成环，直接跳过（冗余边，不选）
            if (uf.connected(X, Y)) {
                continue;
            }

            // 如果X和Y不连通 → 合并它们的连通分量（相当于修这条道路）
            uf.unite(X, Y);
            // 更新最大边长度：因为按升序排序，每次合并的边都是当前最长的
            max_fuel = max(max_fuel, C);

            // 🔹 优化：如果所有城市已经连通（连通分量数为1），提前退出循环，不用遍历剩余边
            if (uf.getCount() == 1) {
                break;
            }
        }

        // 🔹 输出当前测试用例的结果：最小油箱容量
        cout << max_fuel << endl;
    }

    return 0;
}
