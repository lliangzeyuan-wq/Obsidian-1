#include <cstdio>
#include <cstring>
#include <algorithm>
using namespace std;

// 定义数组最大长度，1000010 足够处理大范围数据
const int N = 1000010;

// n: 数组长度, m: 查询/操作次数
int n, m;

// a[N]: 存储原数组元素 (本题中a[i]其实是待插入的值)
// tr[N]: 树状数组的核心结构，用于维护前缀和
int a[N], tr[N];

// 1. 核心函数：lowbit
// 作用：获取数字 x 的二进制表示中，最低位的 1 及其后面的所有 0 构成的数值。
// 例如：x=6(110), lowbit=2(10); x=4(100), lowbit=4(100)
int lowbit(int x)
{
    // 利用补码特性，x & -x 可以快速提取出最低位的 1
    return x & -x;
}

// 2. 操作：单点更新（加法）
// 作用：给数组中第 x 个位置的数值加上 v (或者将 v 赋值到该位置，视场景而定，这里是累加)
// 参数：x - 操作位置, v - 增加的值
void add(int x, int v)
{
    // 从 x 开始，不断向上更新父节点，直到覆盖整个范围 n
    for (int i = x; i <= n; i += lowbit(i))
    {
        tr[i] += v;
    }
}

// 3. 查询：前缀和查询
// 作用：查询从数组第 1 个位置到第 x 个位置的所有数值的总和
// 参数：x - 结束位置
// 返回值：[1...x] 的和
int query(int x)
{
    int res = 0;
    // 从 x 开始，不断跳着向左找，累加所有管辖区间的和
    for (int i = x; i > 0; i -= lowbit(i))
    {
        res += tr[i];
    }
    return res;
}

// 4. 主逻辑
int main()
{
    // 输入 n (数组长度) 和 m (操作次数)
    scanf("%d%d", &n, &m);
    
    // 第一遍循环：读入初始数组 a
    // 注意：这里如果是初始化，通常 a[i] 是原值，然后 add(i, a[i]) 把初始值塞进去
    for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
    
    // 初始化树状数组，将原数组 a 的值逐个插入
    for (int i = 1; i <= n; i++) add(i, a[i]);

    // 处理 m 次查询或修改
    while (m -- )
    {
        int k, x, y;
        // k=0 表示查询操作，k=1 表示更新操作 (根据题目具体定义)
        scanf("%d%d%d", &k, &x, &y);
        
        // 情况1：查询区间 [x, y] 的和
        // 公式：前缀和(1~y) - 前缀和(1~x-1) = 区间和(x~y)
        if (k == 0) 
            printf("%d\n", query(y) - query(x - 1));
        
        // 情况2：对第 x 个位置进行更新 (加上 y，通常 y=1 表示激活，y=-1 表示撤销，视题意)
        // 也可理解为：把第 x 个数的值修改为 y (如果题目是直接赋值，逻辑需调整)
        else 
            add(x, y);
    }
    
    return 0;
}

### 💡 重点提示（结合你之前的 LIS 问题）

你现在学的**最长不下降子序列（LNDS）**的**线段树 / 树状数组**解法，核心就是把这里的 `求和` 改成了 `求最大值`。

1. **普通树状数组**：
    
    - `add`：累加值。
    - `query`：求前缀和。
    
2. **LNDS 专用树状数组**：
    
    - **修改操作**：把 `tr[i] += v` 改成 `tr[i] = max(tr[i], v)`（取最大值）。
    - **查询操作**：把 `res += tr[i]` 改成 `res = max(res, tr[i])`（取最大值）。
