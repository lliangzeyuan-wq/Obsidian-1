int max(string s,int k)
{
	int ans=0,vowel=0;
	int n=s.size();
	for(int i=0;i<n;++i)
	{
			if(s[i]=='a'||s[i]=='e'||s[i]=='i'||s[i]=='o'||s[i]=='u')
		{
			vowel++;
		}
		int r=i-k+1;
		if(r<0)
		{
			continue;
		}
		ans=max(ans,vowel);
		if(s[r]=='a'||s[r]=='e'||s[r]=='i'||s[r]=='o'||s[r]=='u')
		{
			vowel--；
		}
	}
	return ans;
}
