class Solution
{
public:
	
	//反转链表
	ListNode*reverseList(ListNode*head)
	{
		ListNode*prev=nullptr;
		ListNode*curr=head;
		while(curr!=nullptr)
		{
			ListNode*nextTemp=curr->next;
			curr->next=prev;
			prev=curr;
			curr=nextTemp;
		}
		return prev;
	}
	ListNode*endOfFirstHalf(ListNode*head)
	{
		ListNode*fast=head;
		ListNode*slow=head;
		while(fast->next!=nullptr&&fast->next->next!=nullptr)
		{
			fast=fast->next->next;
			slow=slow->next;
		}
		return slow;
	}
}
