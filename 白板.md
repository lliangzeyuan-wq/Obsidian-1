#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

#define BUF_SIZE 1024    // General buffer size for data reception
#define OPSZ 4           // Size of each integer operand (4 bytes)

// Function prototypes
void error_handling(char *message);
int calculate(int opnum, int opnds[], char operator);

int main(int argc, char *argv[])
{
    int serv_sock, clnt_sock;       // Server listening socket, client connection socket
    char opinfo[BUF_SIZE];         // Buffer to store the full operation request
    int result, opnd_cnt, i;       // Calculation result, number of operands, loop index
    int recv_cnt, recv_len;         // Bytes received in one read, total bytes received so far
    struct sockaddr_in serv_adr, clnt_adr;  // Server and client address structures
    socklen_t clnt_adr_sz;          // Size of client address structure

    // Step 1: Validate command-line arguments (must provide port number)
    if(argc != 2)
    {
        printf("Usage : %s <port>\n", argv[0]);
        exit(1);
    }

    // Step 2: Create TCP listening socket
    serv_sock = socket(PF_INET, SOCK_STREAM, 0);
    if(serv_sock == -1)
        error_handling("socket() error");

    // Step 3: Initialize server address structure
    memset(&serv_adr, 0, sizeof(serv_adr));
    serv_adr.sin_family = AF_INET;                  // IPv4 address family
    serv_adr.sin_addr.s_addr = htonl(INADDR_ANY);   // Bind to all available network interfaces
    serv_adr.sin_port = htons(atoi(argv[1]));        // Convert port to network byte order

    // Step 4: Bind socket to the specified IP and port
    if(bind(serv_sock, (struct sockaddr*)&serv_adr, sizeof(serv_adr)) == -1)
        error_handling("bind() error");

    // Step 5: Start listening for client connections (backlog = 5)
    if(listen(serv_sock, 5) == -1)
        error_handling("listen() error");

    clnt_adr_sz = sizeof(clnt_adr);

    // Step 6: Accept up to 5 client connections (loop runs 5 times)
    for(i = 0; i < 5; i++)
    {
        opnd_cnt = 0;

        // Accept a new client connection
        clnt_sock = accept(serv_sock, (struct sockaddr*)&clnt_adr, &clnt_adr_sz);

        // Step 7: First read - get the number of operands (1 byte)
        read(clnt_sock, &opnd_cnt, 1);

        recv_len = 0;

        // Step 8: Read operands and operator until full request is received
        // Expected total length: 1 (count) + opnd_cnt*4 (operands) + 1 (operator)
        while((opnd_cnt * OPSZ + 1) > recv_len)
        {
            // Read remaining data into opinfo buffer
            recv_cnt = read(clnt_sock, &opinfo[recv_len], BUF_SIZE-1);
            recv_len += recv_cnt;
        }

        // Step 9: Perform the calculation
        // opinfo contains operands, last byte is the operator
        result = calculate(opnd_cnt, (int*)opinfo, opinfo[recv_len-1]);

        // Step 10: Send the 4-byte integer result back to the client
        write(clnt_sock, (char*)&result, sizeof(result));

        // Close the client connection
        close(clnt_sock);
    }

    // Close the server listening socket
    close(serv_sock);
    return 0;
}

// Calculation function: performs +, -, * operations
int calculate(int opnum, int opnds[], char op)
{
    int result = opnds[0], i;

    switch(op)
    {
        case '+':
            for(i = 1; i < opnum; i++) result += opnds[i];
            break;
        case '-':
            for(i = 1; i < opnum; i++) result -= opnds[i];
            break;
        case '*':
            for(i = 1; i < opnum; i++) result *= opnds[i];
            break;
    }
    return result;
}

// Error handling function
void error_handling(char *message)
{
    fputs(message, stderr);
    fputc('\n', stderr);
    exit(1);
}
