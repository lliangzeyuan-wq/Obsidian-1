#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    long long S;
    cin >> n >> S;

    vector<pair<int, int>> soldiers(n);
    for (int i = 0; i < n; ++i) {
        int p, c;
        cin >> p >> c;
        soldiers[i] = {c, p}; // 存储为 (训练次数, 单次成本)，便于排序
    }

    // 按训练次数升序排序
    sort(soldiers.begin(), soldiers.end());

    // 预处理后缀和：sum_pi[i] 表示从第i个士兵到最后的单次成本和
    // sum_ci_pi[i] 表示从第i个士兵到最后的 (训练次数 × 单次成本) 和
    vector<long long> sum_pi(n + 1, 0);
    vector<long long> sum_ci_pi(n + 1, 0);
    for (int i = n - 1; i >= 0; --i) {
        sum_pi[i] = sum_pi[i + 1] + soldiers[i].second;
        sum_ci_pi[i] = sum_ci_pi[i + 1] + (long long)soldiers[i].first * soldiers[i].second;
    }

    long long min_cost = LLONG_MAX;
    int prev_t = -1;

    // 遍历所有关键 t 值：0 和所有 c_i
    for (int i = 0; i <= n; ++i) {
        int t;
        if (i == 0) {
            t = 0;
        } else {
            t = soldiers[i - 1].first;
        }

        // 跳过重复的 t 值，避免重复计算
        if (t == prev_t) {
            continue;
        }
        prev_t = t;

        // 找到第一个训练次数 > t 的士兵位置
        auto it = upper_bound(soldiers.begin(), soldiers.end(), make_pair(t, INT_MAX));
        int pos = it - soldiers.begin();

        // 计算当前 t 对应的总花费
        long long current = (long long)t * S + (sum_ci_pi[pos] - (long long)t * sum_pi[pos]);
        if (current < min_cost) {
            min_cost = current;
        }
    }

    cout << min_cost << endl;

    return 0;
}
