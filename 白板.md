#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

struct TrieNode {
    unordered_map<int, TrieNode*> child;
};

TrieNode* root = new TrieNode();
int ans = 0;

vector<int> tree1[105], tree2[105];
int val1[105], val2[105];

void insert(vector<int>& path) {
    TrieNode* cur = root;
    for (int x : path) {
        if (cur->child.find(x) == cur->child.end())
            cur->child[x] = new TrieNode();
        cur = cur->child[x];
    }
}

void query(vector<int>& path) {
    TrieNode* cur = root;
    int len = 0;
    for (int x : path) {
        if (cur->child.find(x) != cur->child.end()) {
            len++;
            cur = cur->child[x];
        }
        else {
            break;
        }
    }
    ans = max(ans, len);
}

void dfs1(int u, int fa, vector<int>& now) {
    now.push_back(val1[u]);
    bool is_leaf = true;
    for (int v : tree1[u]) {
        if (v != fa) {
            is_leaf = false;
            dfs1(v, u, now);
        }
    }
    if (is_leaf) {
        insert(now);
    }
    now.pop_back();
}

void dfs2(int u, int fa, vector<int>& now) {
    now.push_back(val2[u]);
    bool is_leaf = true;
    for (int v : tree2[u]) {
        if (v != fa) {
            is_leaf = false;
            dfs2(v, u, now);
        }
    }
    if (is_leaf) {
        query(now);
    }
    now.pop_back();
}

int main() {
    int n, m;
    cin >> n >> m;

    // 读取第一棵树的权值
    for (int i = 1; i <= n; i++) {
        cin >> val1[i];
    }
    // 读取第二棵树的权值
    for (int i = 1; i <= m; i++) {
        cin >> val2[i];
    }

    // 读取第一棵树的边
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        tree1[u].push_back(v);
        tree1[v].push_back(u);
    }

    // 读取第二棵树的边
    for (int i = 0; i < m - 1; i++) {
        int u, v;
        cin >> u >> v;
        tree2[u].push_back(v);
        tree2[v].push_back(u);
    }

    vector<int> tmp;
    dfs1(1, -1, tmp);
    tmp.clear();
    dfs2(1, -1, tmp);

    cout << ans << endl;
    return 0;
}
