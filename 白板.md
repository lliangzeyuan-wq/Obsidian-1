```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
#include <iomanip>

using namespace std;

const double PI = acos(-1.0);  // 定义圆周率

int main() {
    // 加速输入输出（比赛必备）
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;
    vector<pair<double, double>> intervals;  // 存储危险角度区间 [左, 右]

    for (int i = 0; i < n; ++i) {
        int x, y, r;
        cin >> x >> y >> r;

        // 计算圆心的极角
        double alpha = atan2(y, x);
        // 原点到圆心的距离
        double d = hypot(x, y);
        // 切线夹角
        double theta = asin(r / d);

        // 危险区间
        double l = alpha - theta;
        double r_angle = alpha + theta;

        // 限制在第一象限 [0, π/2]
        l = max(0.0, l);
        r_angle = min(PI / 2, r_angle);

        // 有效区间才加入
        if (l < r_angle) {
            intervals.emplace_back(l, r_angle);
        }
    }

    // 按区间左端点排序
    sort(intervals.begin(), intervals.end());

    double total_danger = 0.0;
    if (!intervals.empty()) {
        double cur_l = intervals[0].first;
        double cur_r = intervals[0].second;

        for (size_t i = 1; i < intervals.size(); ++i) {
            double l = intervals[i].first;
            double r = intervals[i].second;
            if (l <= cur_r) {
                // 合并区间
                cur_r = max(cur_r, r);
            } else {
                // 累加危险长度
                total_danger += cur_r - cur_l;
                cur_l = l;
                cur_r = r;
            }
        }
        // 累加最后一个区间
        total_danger += cur_r - cur_l;
    }

    // 计算安全概率
    double total_angle = PI / 2;
    double prob = (total_angle - total_danger) / total_angle;

    // 保留三位小数输出
    cout << fixed << setprecision(3) << prob << endl;

    return 0;
}
```

# 代码说明

1. **核心逻辑**
    
    - 几何计算：极角、切线夹角、危险区间裁剪（仅保留第一象限）
    - 区间合并：排序后遍历合并重叠区间，计算总危险角度
    - 概率计算：安全角度 / 总角度（π/2）
    
2. **比赛优化**
    
    - `ios::sync_with_stdio(false); cin.tie(nullptr);` 加速输入输出（处理大数据不超时）
    - 用 `vector` 存储区间，排序效率高
    - 用 `acos(-1.0)` 精准定义圆周率，避免精度误差
    
3. **输出格式**
    
    - `fixed << setprecision(3)` 严格保留三位小数，符合题目要求
