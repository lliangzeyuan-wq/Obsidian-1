#include<bits/stdc++.h>
using namespace std;

const int N = 100010;
const int LOG = 20;      // 倍增层数
const int FOOD = 21;    // 零食种类 1~20

// ---------------------- 树结构 ----------------------
vector<int> g[N];       // 邻接表
int c[N];               // 每个节点的零食

// ---------------------- LCA 倍增 ----------------------
int up[N][LOG];        // up[u][k] = u 跳 2^k 步到哪
int depth[N];          // 节点深度

// ---------------------- 核心计数 ----------------------
int cnt[N][FOOD];      // cnt[u][x] = 根到u，零食x出现多少次


// ---------------------- DFS 初始化 ----------------------
void dfs(int u, int fa) {
    // 1. 记录父节点
    up[u][0] = fa;
    depth[u] = depth[fa] + 1;

    // 2. 继承父节点的零食数量
    for (int x = 1; x <= 20; x++) {
        cnt[u][x] = cnt[fa][x];
    }

    // 3. 加上当前节点的零食
    cnt[u][c[u]]++;

    // 4. 预处理倍增表
    for (int k = 1; k < LOG; k++) {
        up[u][k] = up[ up[u][k-1] ][k-1];
    }

    // 5. 遍历子节点
    for (int v : g[u]) {
        if (v != fa) dfs(v, u);
    }
}

// ---------------------- LCA 板子 ----------------------
int lca(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);

    // 把 u 提到和 v 同深度
    for (int k = LOG-1; k >= 0; k--) {
        if (depth[up[u][k]] >= depth[v]) {
            u = up[u][k];
        }
    }

    if (u == v) return u;

    // 一起往上跳
    for (int k = LOG-1; k >= 0; k--) {
        if (up[u][k] != up[v][k]) {
            u = up[u][k];
            v = up[v][k];
        }
    }

    return up[u][0];
}

// ---------------------- 主函数 ----------------------
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n, q;
    cin >> n >> q;

    // 读零食
    for (int i = 1; i <= n; i++) {
        cin >> c[i];
    }

    // 建树
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    // DFS 预处理
    dfs(1, 0);

    // 处理询问
    while (q--) {
        int u, v;
        cin >> u >> v;
        int L = lca(u, v);

        int ans = 0;
        for (int x = 1; x <= 20; x++) {
            // 路径 u->v 的零食数量
            int have = cnt[u][x] + cnt[v][x] - 2 * cnt[L][x];
            
            // LCA 被减了两次，要加回来
            if (c[L] == x) have++;

            if (have > 0) ans++;
        }

        cout << ans << '\n';
    }

    return 0;
}
