- 类模板中成员函数类外实现时，需要加上模板参数列表
```
#include<iostream>
using namespace std;
#include<string>
template <class T1, class T2>
class Person {
public:
	T1 m_Name;
	T2 m_Age;
	Person(T1 name, T2 age);
	void showPerson();
};
//类外实现

//构造函数的类外实现
template<class T1,class T2>
Person<T1,T2>::Person(T1 name, T2 age) {
	m_Name = name;
	m_Age = age;
}
//成员函数的类外实现
template<class T1,class T2>
void Person<T1,T2>::showPerson() {
	cout << "name:" << this->m_Name << " age:" << this->m_Age << endl;
}
int main() {
	system("pause");
	return 0;
}
```
